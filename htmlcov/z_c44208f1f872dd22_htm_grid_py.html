11class HTMBuilder(BaseGridBuilder): 

12 """Hierarchical Triangular Mesh (HTM) grid. 

13 

14 HTM divides the sphere into an octahedron (8 triangular faces), then 

15 recursively subdivides each face into 4 smaller triangles by inserting 

16 edge-midpoint vertices projected onto the unit sphere. The recursion 

17 depth is controlled by ``htm_level``. This produces a strictly 

18 hierarchical triangulation: every triangle at level *n* is the union of 

19 exactly 4 triangles at level *n* + 1. 

20 

21 Cell areas are *approximately* equal but not strictly so — area 

22 uniformity improves with level because the icosahedral edge-length 

23 asymmetry averages out over many subdivisions. 

24 

25 Coordinate convention (physics / GNSS) 

26 --------------------------------------- 

27 * phi ∈ [0, 2π) – azimuthal angle from North, clockwise (navigation convention) 

28 * theta ∈ [0, π/2] – polar angle from zenith (0 = straight up, 

29 π/2 = horizon) 

30 

31 Cell centers are the 3D Cartesian mean of the three triangle vertices, 

32 re-normalized onto the unit sphere. 

33 

34 What ``htm_level`` (resolution) means 

35 -------------------------------------- 

36 The resolution is set by ``htm_level``, **not** by ``angular_resolution``. 

37 ``angular_resolution`` is used only to *estimate* an appropriate level 

38 when ``htm_level`` is not supplied explicitly. The heuristic is:: 

39 

40 target_edge ≈ 2 × angular_resolution (degrees) 

41 htm_level = min(15, ceil(log₂(90 / target_edge))) 

42 

43 The approximate triangle edge length at level *n* is:: 

44 

45 edge ≈ 90° / 2ⁿ 

46 

47 | Level | Triangles (full sphere) | Approx edge | 

48 |-------|-------------------------|-------------| 

49 | 0 | 8 | 90° | 

50 | 1 | 32 | 45° | 

51 | 2 | 128 | 22.5° | 

52 | 3 | 512 | 11.25° | 

53 | 4 | 2 048 | 5.6° | 

54 | n | 8 × 4ⁿ | 90° / 2ⁿ | 

55 

56 Mathematical construction 

57 ------------------------- 

58 1. **Octahedron** – 6 vertices at ±x, ±y, ±z on the unit sphere, forming 

59 8 triangular faces (4 northern, 4 southern). 

60 2. **Subdivision** – for each triangle [v₀, v₁, v₂], three edge 

61 midpoints are computed and projected onto the unit sphere:: 

62 

63 m₀ = normalize((v₀ + v₁) / 2) 

64 m₁ = normalize((v₁ + v₂) / 2) 

65 m₂ = normalize((v₂ + v₀) / 2) 

66 

67 The four children are [v₀, m₀, m₂], [v₁, m₁, m₀], [v₂, m₂, m₁], 

68 and [m₀, m₁, m₂]. This is repeated ``htm_level`` times. 

69 3. **Hemisphere filter** – a triangle is kept if its *centroid* (the 

70 3D Cartesian mean of the three vertices, renormalised to the unit 

71 sphere) satisfies ``theta ≤ π/2 − cutoff_theta``. Filtering on 

72 centroid rather than "any vertex above" avoids keeping boundary 

73 triangles that are mostly below the horizon; kept triangles may 

74 still extend slightly across the horizon since cells are not 

75 clipped to an exact circle. 

76 4. Each leaf triangle becomes one cell; its center, bounding box, and 

77 three vertex coordinates are stored. 

78 

79 Parameters 

80 ---------- 

81 angular_resolution : float 

82 Approximate angular resolution in degrees. Used only to derive 

83 ``htm_level`` when that parameter is not given explicitly. 

84 cutoff_theta : float 

85 Elevation mask angle in degrees. Triangles are excluded if their 

86 centroid is below this elevation. 

87 htm_level : int or None 

88 HTM subdivision depth. If ``None``, estimated from 

89 ``angular_resolution``. Practical range 0–15. 

90 phi_rotation : float 

91 Rigid azimuthal rotation applied after construction, in degrees. 

92 

93 Notes 

94 ----- 

95 The ``theta_lims``, ``phi_lims``, and ``cell_ids`` fields of the returned 

96 ``GridData`` are *synthetic* evenly-spaced arrays kept only for interface 

97 compatibility with ring-based grids. They do **not** describe the actual 

98 triangular cell layout. 

99 

100 HTM IDs in this implementation use a decimal-digit scheme 

101 (``parent_id × 10 + child_index``) which diverges from the original 

102 SDSS HTM binary-coded ID scheme. This is adequate for indexing but 

103 should not be compared with external HTM catalogs. 

104 

105 References 

106 ---------- 

107 Kunszt et al. (2001): "The Hierarchical Triangular Mesh" 

108 https://www.sdss.org/dr12/algorithms/htm/ 

109 

110 """

160 def _build_grid( 

161 self, 

162 ) -> tuple[pl.DataFrame, np.ndarray, list[np.ndarray], list[np.ndarray]]: 

163 """Build HTM grid by recursive octahedron subdivision. 

164 

165 Returns 

166 ------- 

167 grid : pl.DataFrame 

168 One row per triangular cell. Contains phi, theta (center), 

169 bounding-box limits, ``htm_id``, ``htm_level``, and the three 

170 vertex coordinate columns ``htm_vertex_0/1/2`` (each a list of 

171 3 floats in Cartesian xyz). 

172 theta_lims : np.ndarray 

173 Synthetic evenly-spaced theta limits (interface compatibility only). 

174 phi_lims : list[np.ndarray] 

175 Synthetic evenly-spaced phi limits (interface compatibility only). 

176 cell_ids : list[np.ndarray] 

177 Single-element list containing all cell ids. 

178 

179 """ 

180 base_vertices = np.array( 

181 [ 

182 [0, 0, 1], # 0: North pole 

183 [1, 0, 0], # 1: +X 

184 [0, 1, 0], # 2: +Y 

185 [-1, 0, 0], # 3: -X 

186 [0, -1, 0], # 4: -Y 

187 [0, 0, -1], # 5: South pole 

188 ], 

189 dtype=np.float64, 

190 ) 

191 

192 base_faces = [ 

193 [0, 1, 2], 

194 [0, 2, 3], 

195 [0, 3, 4], 

196 [0, 4, 1], # Northern 

197 [5, 2, 1], 

198 [5, 3, 2], 

199 [5, 4, 3], 

200 [5, 1, 4], # Southern 

201 ] 

202 

203 all_triangles = [] 

204 all_htm_ids = [] 

205 

206 for base_idx, base_face in enumerate(base_faces): 

207 v0 = base_vertices[base_face[0]] 

208 v1 = base_vertices[base_face[1]] 

209 v2 = base_vertices[base_face[2]] 

210 

211 triangles, ids = self._subdivide_htm([v0, v1, v2], base_idx, self.htm_level) 

212 all_triangles.extend(triangles) 

213 all_htm_ids.extend(ids) 

214 

215 max_theta = np.pi / 2 - self.cutoff_theta_rad 

216 

217 # Convert to cells 

218 cells = [] 

219 for tri, htm_id in zip(all_triangles, all_htm_ids): 

220 v0, v1, v2 = tri 

221 

222 # Center 

223 center = (v0 + v1 + v2) / 3 

224 center = center / np.linalg.norm(center) 

225 

226 theta_center = np.arccos(np.clip(center[2], -1, 1)) 

227 

228 # Filter hemisphere on the centroid, not "any vertex above" -- 

229 # the latter keeps boundary triangles that are mostly below the 

230 # horizon. 

231 if theta_center > max_theta: 

232 continue 

233 

234 phi_center = np.arctan2(center[1], center[0]) 

235 phi_center = np.mod(phi_center, 2 * np.pi) 

236 

237 # Vertex coords 

238 thetas, phis = [], [] 

239 for v in [v0, v1, v2]: 

240 t = np.arccos(np.clip(v[2], -1, 1)) 

241 p = np.arctan2(v[1], v[0]) 

242 p = np.mod(p, 2 * np.pi) 

243 thetas.append(t) 

244 phis.append(p) 

245 

246 phi_min, phi_max = phi_bbox(np.array(phis)) 

247 

248 cells.append( 

249 { 

250 "phi": phi_center, 

251 "theta": theta_center, 

252 "phi_min": phi_min, 

253 "phi_max": phi_max, 

254 "theta_min": min(thetas), 

255 "theta_max": max(thetas), 

256 "htm_id": htm_id, 

257 "htm_level": self.htm_level, 

258 "htm_vertex_0": v0.tolist(), 

259 "htm_vertex_1": v1.tolist(), 

260 "htm_vertex_2": v2.tolist(), 

261 } 

262 ) 

263 

264 grid = pl.DataFrame(cells).with_columns( 

265 pl.int_range(0, pl.len()).alias("cell_id") 

266 ) 

267 

268 theta_lims = np.linspace(0, np.pi / 2, 10) 

269 phi_lims = [np.linspace(0, 2 * np.pi, 20) for _ in range(len(theta_lims))] 

270 cell_ids_list = [grid["cell_id"].to_numpy()] 

271 

272 return grid, theta_lims, phi_lims, cell_ids_list

274 def _subdivide_htm( 

275 self, 

276 tri: list, 

277 htm_id: int, 

278 target_level: int, 

279 current_level: int = 0, 

280 ) -> tuple[list, list]: 

281 """Recursively subdivide a single triangle. 

282 

283 Parameters 

284 ---------- 

285 tri : list of np.ndarray 

286 Three vertex arrays [v₀, v₁, v₂], each shape ``(3,)``. 

287 htm_id : int 

288 Current HTM identifier for this triangle. 

289 target_level : int 

290 Recursion depth to reach. 

291 current_level : int 

292 Current recursion depth. 

293 

294 Returns 

295 ------- 

296 triangles : list of list 

297 Leaf triangles at ``target_level``. 

298 ids : list of int 

299 Corresponding HTM identifiers. 

300 

301 """ 

302 if current_level == target_level: 

303 return [tri], [htm_id] 

304 

305 v0, v1, v2 = tri 

306 

307 # Midpoints on sphere 

308 m0 = (v0 + v1) / 2 

309 m0 = m0 / np.linalg.norm(m0) 

310 m1 = (v1 + v2) / 2 

311 m1 = m1 / np.linalg.norm(m1) 

312 m2 = (v2 + v0) / 2 

313 m2 = m2 / np.linalg.norm(m2) 

314 

315 # 4 children 

316 children = [[v0, m0, m2], [v1, m1, m0], [v2, m2, m1], [m0, m1, m2]] 

317 

318 all_tris = [] 

319 all_ids = [] 

320 

321 for child_idx, child in enumerate(children): 

322 child_id = htm_id * 10 + child_idx 

323 tris, ids = self._subdivide_htm( 

324 child, 

325 child_id, 

326 target_level, 

327 current_level + 1, 

328 ) 

329 all_tris.extend(tris) 

330 all_ids.extend(ids) 

331 

332 return all_tris, all_ids
