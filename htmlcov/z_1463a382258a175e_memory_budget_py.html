1"""The memory budget of a run: how many workers and days per unit fit. 

2 

3``processing.params.max_memory_gb`` is the budget; without it, the memory 

4available when the run starts. The budget holds the main process and the 

5workers: 

6 

7- The main process holds what it holds when the run starts, plus the data 

8 of the days in flight: two units of days, the one going to VOD and the 

9 one being read (:func:`~canvodpy.orchestrator.pipeline._run_ahead`). 

10- Each worker holds the Python libraries, plus its largest task: a task 

11 reads the files that share an epoch chunk, one after the other, and 

12 holds their data until it writes them. With ``processing.preprocessing`` 

13 set the chunks are counted at the aggregation interval, so the files of 

14 a time bin are in one task. 

15 

16Workers = min(the worker cap, (budget - main process with one day) / 

17largest task). A unit then gets as many days as the rest of the budget 

18holds, at most :data:`MAX_UNIT_DAYS` (ADR 0021). A run in which not even 

19one worker fits with one day stops before reading a file 

20(:class:`MemoryBudgetError`). The estimates come from the factors below, 

21measured on the test data; the workers log their real peaks 

22(``worker_memory_peak``, :func:`~canvodpy.orchestrator.resources.run_measured`). 

23"""

32GB = 1024**3

34#: Memory of a worker before its first task: Python and the scientific 

35#: libraries (0.24 GB measured on macOS, up to 0.33 GB after tasks). 

36WORKER_BASE_BYTES = int(0.35 * GB) 

37 

38#: Peak memory of a worker task per byte of its input files, by reader. 

39#: Reading alone peaks at 66 (RINEX v2.11), 28 (RINEX v3.04), 15 (SBF) and 

40#: 8 (NMEA) times the file size on the test data; the factors add half of 

41#: that for the coordinates and the write. 

42TASK_BYTES_PER_INPUT_BYTE: Mapping[str, float] = { 

43 "rinex2": 100.0, 

44 "rinex3": 45.0, 

45 "rinex3_stripped": 45.0, 

46 "sbf": 25.0, 

47 "nmea": 15.0, 

48} 

49 

50#: Memory a task holds per byte of its input files once they are read, until 

51#: it writes them. Measured on a receiver-day of 96 RINEX v3 files of 15 

52#: minutes (208 MB, SNR only, preprocessing on): about 1.6, which with the 

53#: largest file's read peak gives the 0.78 GB peak measured for the task (#417). 

54TASK_HELD_BYTES_PER_INPUT_BYTE = 1.6 

55 

56#: For a reader not listed above: the largest factor. 

57UNKNOWN_READER_FACTOR = max(TASK_BYTES_PER_INPUT_BYTE.values()) 

58 

59#: Memory the main process holds per byte of a day's input files while the 

60#: day is in flight, by reader (#452). Measured on the test data as the 

61#: growth of the main process's peak from half a day to a whole day, three 

62#: runs each, SNR only: 0.12 for SBF (median; at most 0.21), from 1.2 GB 

63#: more input. The peak of identical runs varies by about 0.1 GB, as much 

64#: as RINEX and NMEA days grow (15 to 190 MB more input), so their 

65#: factors stay at 1.0 for RINEX v3 (0.45 measured before) and 2.5 for 

66#: RINEX v2.11 and NMEA (medians 2.1 and 2.4, small files). 

67MAIN_BYTES_PER_INPUT_BYTE: Mapping[str, float] = { 

68 "rinex2": 2.5, 

69 "rinex3": 1.0, 

70 "rinex3_stripped": 1.0, 

71 "sbf": 0.25, 

72 "nmea": 2.5, 

73} 

74 

75#: For a reader not listed in :data:`MAIN_BYTES_PER_INPUT_BYTE`: the largest 

76#: factor. 

77UNKNOWN_READER_MAIN_FACTOR = max(MAIN_BYTES_PER_INPUT_BYTE.values()) 

78 

79#: Units whose data are in memory at the same time (#277). 

80UNITS_IN_MEMORY = 2 

81 

82#: Most days of a unit, so that a lost conflict or a crash loses little 

83#: (ADR 0021). 

84MAX_UNIT_DAYS = 31

115 def log_fields(self) -> dict[str, object]: 

116 """The plan for the log, in GB. 

117 

118 ``unit_limit`` says what set the days per unit: ``"memory"``, 

119 ``"max_unit_days"`` (:data:`MAX_UNIT_DAYS`), ``"days_to_run"`` 

120 (the run has no more days) or ``"days_per_batch"`` (the 

121 deprecated setting). 

122 """ 

123 return { 

124 "budget_gb": round(self.budget_bytes / GB, 2), 

125 "main_process_gb": round(self.main_bytes / GB, 2), 

126 "day_gb": round(self.day_bytes / GB, 2), 

127 "task_peak_gb": round(self.task_peak_bytes / GB, 2), 

128 "largest_task": [p.name for p in self.largest_task], 

129 "workers": self.workers, 

130 "unit_days": self.unit_days, 

131 "unit_limit": self.unit_limit, 

132 "tasks_per_day": round(self.tasks_per_day, 1), 

133 "tasks_per_worker": round( 

134 self.unit_days * self.tasks_per_day / self.workers, 1 

135 ), 

136 }

139def main_day_bytes(files: Iterable[InputFile]) -> int: 

140 """Estimated memory the main process holds for a day of ``files``.""" 

141 return int( 

142 sum( 

143 f.size * MAIN_BYTES_PER_INPUT_BYTE.get(f.reader, UNKNOWN_READER_MAIN_FACTOR) 

144 for f in files 

145 ) 

146 )

148 

149def task_peak_bytes(files: Sequence[InputFile]) -> int: 

150 """Estimated peak memory of a worker that reads ``files`` in one task. 

151 

152 The task reads the files one after the other and holds their data until 

153 it writes them: the peak of reading its largest file, plus the data of 

154 all of them. 

155 """ 

156 if not files: 

157 return WORKER_BASE_BYTES 

158 reading = max( 

159 f.size * TASK_BYTES_PER_INPUT_BYTE.get(f.reader, UNKNOWN_READER_FACTOR) 

160 for f in files 

161 ) 

162 held = sum(f.size for f in files) * TASK_HELD_BYTES_PER_INPUT_BYTE 

163 return WORKER_BASE_BYTES + int(reading + held)

166def chunk_tasks( 

167 files: Sequence[InputFile], 

168 *, 

169 epoch_chunk: int, 

170 cadence_s: float | None, 

171) -> list[list[InputFile]]: 

172 """The files of one receiver and day, grouped as the run's tasks group them. 

173 

174 Files whose names put them into a shared epoch chunk are one task; the 

175 chunks start at 00:00 at the day's ``cadence_s`` (the aggregation 

176 interval, or else the finest sampling the names give). A time bin is 

177 one slot of the aggregated epochs, so the files of a bin share its 

178 chunk. A file without a canonical name is a task of its own. 

179 """ 

180 from canvod.preflight.convention import CanVODFilename 

181 from canvodpy.orchestrator.forked_writes import chunk_groups 

182 

183 named = [ 

184 (f, CanVODFilename.from_filename(f.canonical_name)) 

185 for f in files 

186 if f.canonical_name 

187 ] 

188 if cadence_s is None: 

189 intervals = [n.sampling_interval.total_seconds() for _, n in named] 

190 cadence_s = min(intervals) if intervals else None 

191 chunks_of: list[tuple[InputFile, set[object]]] = [] 

192 for f in files: 

193 if not f.canonical_name or not cadence_s: 

194 chunks_of.append((f, {("own", f.path)})) 

195 continue 

196 name = CanVODFilename.from_filename(f.canonical_name) 

197 midnight = datetime.combine( 

198 name.start.date(), datetime.min.time(), name.start.tzinfo 

199 ) 

200 chunk_s = epoch_chunk * cadence_s 

201 first = (name.start - midnight).total_seconds() // chunk_s 

202 last_epoch = name.end - name.sampling_interval 

203 last = max(first, (last_epoch - midnight).total_seconds() // chunk_s) 

204 chunks_of.append((f, {int(c) for c in range(int(first), int(last) + 1)})) 

205 return [members for members, _ in chunk_groups(chunks_of)] 

206

208def largest_task(tasks: Iterable[Sequence[InputFile]]) -> Sequence[InputFile]:

210 return max(tasks, key=task_peak_bytes, default=[])

213def plan_memory( 

214 *, 

215 budget_bytes: int, 

216 main_now_bytes: int, 

217 day_bytes: int, 

218 task: Sequence[InputFile], 

219 tasks_per_day: float, 

220 max_workers: int, 

221 days_to_run: int, 

222 budget_source: str, 

223 days_per_batch: int | None = None, 

224) -> MemoryPlan: 

225 """Size the workers and the days per unit to fit ``budget_bytes``. 

226 

227 The workers come first: as many as fit next to the main process with 

228 one day in flight. A unit then gets the most days the rest of the 

229 budget holds, at most :data:`MAX_UNIT_DAYS` and the days the run has 

230 (ADR 0021). 

231 

232 Parameters 

233 ---------- 

234 budget_bytes 

235 The memory the run may use. 

236 main_now_bytes 

237 Memory of the main process when the run starts. 

238 day_bytes 

239 Memory the main process holds for the largest day (all 

240 receivers): :func:`main_day_bytes` of its input files. 

241 task 

242 The largest task (:func:`largest_task`). 

243 tasks_per_day 

244 The tasks of a day, on average over the run's days; for the log. 

245 max_workers 

246 The worker cap the caller passes: the pipeline's 

247 ``n_max_workers``, else ``os.cpu_count()``, else 4. In 

248 ``resource_mode: auto``, ``n_max_workers`` is one per physical 

249 core, at most the logical cores minus two, at least 1. 

250 days_to_run 

251 The days the run processes. 

252 budget_source 

253 Where the budget comes from, for the error message. 

254 days_per_batch 

255 The deprecated ``processing.params.days_per_batch``: if given, a 

256 unit has at most this many days. 

257 

258 Raises 

259 ------ 

260 MemoryBudgetError 

261 If one worker does not fit with one day per unit. 

262 """ 

263 peak = task_peak_bytes(task) 

264 

265 def main_bytes(days: int) -> int: 

266 return main_now_bytes + UNITS_IN_MEMORY * days * day_bytes 

267 

268 if main_bytes(1) + peak > budget_bytes: 

269 files = ", ".join(f.path.name for f in task) 

270 msg = ( 

271 f"The run needs at least {(main_bytes(1) + peak) / GB:.1f} GB of " 

272 f"memory, but its budget is {budget_bytes / GB:.1f} GB " 

273 f"({budget_source}). The main process needs " 

274 f"{main_bytes(1) / GB:.1f} GB with one day per unit " 

275 f"({UNITS_IN_MEMORY} days in flight), and one worker reading " 

276 f"{files} needs {peak / GB:.1f} GB. Raise " 

277 "processing.params.max_memory_gb or free memory." 

278 ) 

279 raise MemoryBudgetError(msg) 

280 workers = int(max(1, min(max_workers, (budget_bytes - main_bytes(1)) // peak))) 

281 

282 limits = {"max_unit_days": MAX_UNIT_DAYS, "days_to_run": max(1, days_to_run)} 

283 if days_per_batch is not None: 

284 limits["days_per_batch"] = max(1, days_per_batch) 

285 unit_limit = min(limits, key=lambda name: limits[name]) 

286 days = limits[unit_limit] 

287 while days > 1 and main_bytes(days) + workers * peak > budget_bytes: 

288 days -= 1 

289 unit_limit = "memory" 

290 return MemoryPlan( 

291 budget_bytes=budget_bytes, 

292 main_bytes=main_bytes(days), 

293 day_bytes=day_bytes, 

294 task_peak_bytes=peak, 

295 largest_task=tuple(f.path for f in task), 

296 workers=workers, 

297 unit_days=days, 

298 unit_limit=unit_limit, 

299 tasks_per_day=tasks_per_day, 

300 )
