1"""Metadata definitions for RINEX datasets. 

2 

3Defines CF-compliant metadata for xarray.Dataset coordinates, data variables, 

4and global attributes. Used by all readers to ensure consistent NetCDF output 

5compatible with downstream VOD calculations and storage operations. 

6 

7Module Contents 

8--------------- 

9OBSERVABLES_METADATA : dict 

10 Metadata for GNSS observables (Pseudorange, Phase, Doppler, etc.) 

11CN0_METADATA : dict 

12 Metadata for Carrier-to-Noise density ratio (C/N0) measurements. 

13SNR_METADATA : dict 

14 Metadata for Signal-to-Noise ratio (SNR) measurements. 

15COORDS_METADATA : dict 

16 Metadata for dataset coordinates (epoch, sid, sv, etc.) 

17DTYPES : dict 

18 Data types for coordinates and variables. 

19DATAVARS_TO_BE_FILLED : dict 

20 Metadata for auxiliary variables (r, theta, phi, v, a). 

21 

22Notes 

23----- 

24All metadata follows CF (Climate and Forecast) conventions for NetCDF files 

25to ensure interoperability with standard tools and libraries. 

26

245# Time scale of the epoch coordinate, recorded in its "time_system" 

246# attribute. Every stored epoch is GPS time, so that all receivers and the 

247# orbits and clocks (SP3, CLK) share one time scale. RINEX and SBF are 

248# written in GPS time; the NMEA reader converts its UTC times to GPS time. 

249# Identifiers follow RINEX 3.04 Table A2 (TIME OF FIRST OBS). 

250EPOCH_TIME_SYSTEM: Final = "GPS" 

251EPOCH_TIME_SYSTEM_NAME: Final = "GPS time"

254def require_gps_time(time_system: str, source: str | None = None) -> None: 

255 """Refuse epochs that are not in GPS time. 

256 

257 Parameters 

258 ---------- 

259 time_system : str 

260 Time scale of the epochs (RINEX 3.04 Table A2 identifier). 

261 source : str, optional 

262 The file the epochs come from, named in the error. 

263 

264 Raises 

265 ------ 

266 ValueError 

267 If ``time_system`` is not ``"GPS"``. 

268 

269 """ 

270 if time_system != EPOCH_TIME_SYSTEM: 

271 where = f"{source}: " if source else "" 

272 msg = ( 

273 f"{where}epochs are in time system {time_system!r}, but canVODpy " 

274 "stores every epoch in GPS time ('GPS'), the time scale of the " 

275 "orbits and clocks, and does not convert from " 

276 f"{time_system!r}. Convert the epochs to GPS time first." 

277 ) 

278 raise ValueError(msg)

326def get_global_attrs() -> dict[str, str]: 

327 """Build global attributes from the active configuration. 

328 

329 Returns 

330 ------- 

331 dict[str, str] 

332 Global attributes dict suitable for xarray Dataset attrs. 

333 Falls back to ``{"Software": "canVODpy", "Institution": "Unknown"}`` 

334 when no config is available (e.g. standalone ``canvod-readers`` 

335 install without a ``canvod-settings.yaml``) — "Unknown" satisfies 

336 ``REQUIRED_ATTRS`` (``base.py``) while remaining a recognizable 

337 placeholder rather than a real institution name, so it can't be 

338 mistaken for genuine FAIR/DataCite metadata downstream. 

339 """ 

340 from canvod.config import load_config 

341 

342 try: 

343 cfg = load_config() 

344 meta = cfg.processing.metadata 

345 except Exception: 

346 return {"Software": "canVODpy", "Institution": "Unknown"} 

347 

348 attrs: dict[str, str] = { 

349 "Author": meta.author, 

350 "Email": meta.email, 

351 "Institution": meta.institution, 

352 } 

353 if meta.department: 

354 attrs["Department"] = meta.department 

355 rg_parts = [p for p in [meta.research_group, meta.website] if p] 

356 if rg_parts: 

357 attrs["Research Group"] = ", ".join(rg_parts) 

358 attrs["Software"] = "canVODpy" 

359 return attrs
