1"""Virtual renaming engine: map physical files to canVOD conventional names. 

2 

3The ``FilenameMapper`` discovers files on disk according to the configured 

4directory layout and source pattern, then wraps each in a ``VirtualFile`` 

5that pairs the physical path with its canVOD conventional name. 

6 

7.. deprecated:: 

8 ``FilenameMapper`` is left over from development and will be removed 

9 with the next major version. Naming recipes (``canvod-filemap``) map 

10 non-canonical filenames, and ``canvodpy run`` finds the files in any 

11 folder layout (:mod:`canvodpy.orchestrator.discovery`). 

12"""

64_EXT_TO_FILETYPE: dict[str, FileType] = { 

65 ".rnx": FileType.RNX, 

66 ".crx": FileType.RNX, 

67 ".sbf": FileType.SBF, 

68 ".ubx": FileType.UBX, 

69 ".nmea": FileType.NMEA, 

70 ".obs": FileType.RNX,

73_RNX2_OBS_CHARS = set("oOdD") 

74_COMPRESSION_EXTS = {".zip", ".gz", ".bz2", ".zst", ".Z"}

77def _detect_file_type(path: Path) -> tuple[str, str | None]: 

78 """Detect the file type and compression from a file path.""" 

79 suffixes = path.suffixes 

80 compression = None 

81 

82 if suffixes and suffixes[-1] in _COMPRESSION_EXTS: 

83 compression = suffixes[-1].lstrip(".") 

84 suffixes = suffixes[:-1] 

85 

86 if not suffixes: 

87 stem = path.stem if compression else path.name 

88 if stem.endswith("_"): 

89 return FileType.SBF, compression 

90 raise ValueError(f"Cannot detect file type for {path.name}") 

91 

92 ext = suffixes[-1].lower() 

93 

94 if ext in _EXT_TO_FILETYPE: 

95 return _EXT_TO_FILETYPE[ext], compression 

96 

97 if ext[1:] in known_file_types(): 

98 return ext[1:], compression 

99 

100 if len(ext) == 4 and ext[1:3].isdigit(): 

101 type_char = ext[3] 

102 if type_char in _RNX2_OBS_CHARS: 

103 return FileType.RNX, compression 

104 

105 if ext.endswith("_") and len(ext) == 4 and ext[1:3].isdigit(): 

106 return FileType.SBF, compression 

107 

108 raise ValueError(f"Cannot detect file type for {path.name}")

148 def discover_all(self) -> list[VirtualFile]: 

149 """Discover all files and map them to conventional names.""" 

150 files = self._discover_files() 

151 results = [] 

152 for path in natsorted(files, key=lambda p: p.name): 

153 try: 

154 vf = self.map_single_file(path) 

155 results.append(vf) 

156 except ValueError, KeyError: 

157 logger.warning("Could not map %s; skipping", path.name) 

158 continue 

159 return results

161 def discover_for_date(self, year: int, doy: int) -> list[VirtualFile]: 

162 """Discover files for a specific date.""" 

163 layout = self.receiver_naming.directory_layout 

164 

165 if layout == DirectoryLayout.YYDDD_SUBDIRS: 

166 yy = year % 100 

167 dir_name = f"{yy:02d}{doy:03d}" 

168 search_dir = self.receiver_base_dir / dir_name 

169 elif layout == DirectoryLayout.YYYYDDD_SUBDIRS: 

170 dir_name = f"{year:04d}{doy:03d}" 

171 search_dir = self.receiver_base_dir / dir_name 

172 else: 

173 all_files = self.discover_all() 

174 return [ 

175 vf 

176 for vf in all_files 

177 if vf.conventional_name.year == year and vf.conventional_name.doy == doy 

178 ] 

179 

180 if not search_dir.is_dir(): 

181 return [] 

182 

183 files = self._glob_in_dir(search_dir) 

184 results = [] 

185 for path in natsorted(files, key=lambda p: p.name): 

186 try: 

187 vf = self.map_single_file(path, year=year, doy=doy) 

188 results.append(vf) 

189 except ValueError, KeyError: 

190 logger.warning("Could not map %s; skipping", path.name) 

191 continue 

192 return results

194 def map_single_file( 

195 self, file_path: Path, *, year: int | None = None, doy: int | None = None 

196 ) -> VirtualFile: 

197 """Map a single physical file to its canVOD conventional name. 

198 

199 Raises 

200 ------ 

201 ValueError 

202 If the file cannot be matched or mapped. 

203 """ 

204 filename = file_path.name 

205 pattern_name = self.receiver_naming.source_pattern 

206 

207 result = match_pattern(filename, pattern_name) 

208 if result is None: 

209 raise ValueError(f"No pattern matched for {filename!r}") 

210 

211 pat, m = result 

212 groups = m.groupdict() 

213 

214 expected_station = self.receiver_naming.source_station 

215 if expected_station and "station" in groups: 

216 actual_station = groups["station"] 

217 if actual_station.lower() != expected_station.lower(): 

218 raise ValueError( 

219 f"Station code mismatch for {filename!r}: " 

220 f"expected {expected_station!r}, got {actual_station!r}" 

221 ) 

222 

223 if year is None: 

224 if "year" in groups and groups["year"] is not None: 

225 year = int(groups["year"]) 

226 elif "yy" in groups and groups["yy"] is not None: 

227 year = resolve_year_from_yy(int(groups["yy"])) 

228 else: 

229 raise ValueError(f"Cannot determine year from {filename!r}") 

230 

231 if doy is None: 

232 if "doy" in groups and groups["doy"] is not None: 

233 doy = int(groups["doy"]) 

234 else: 

235 raise ValueError(f"Cannot determine DOY from {filename!r}") 

236 

237 if "hour" in groups and groups["hour"] is not None: 

238 hour = int(groups["hour"]) 

239 elif "hour_letter" in groups and groups["hour_letter"] is not None: 

240 hour = hour_letter_to_int(groups["hour_letter"]) 

241 else: 

242 hour = 0 

243 

244 if "minute" in groups and groups["minute"] is not None: 

245 minute = int(groups["minute"]) 

246 else: 

247 minute = 0 

248 

249 sampling = ( 

250 groups.get("sampling") 

251 or self.receiver_naming.sampling 

252 or self.site_naming.default_sampling 

253 ) 

254 period = groups.get("period") 

255 if not period: 

256 hour_letter = groups.get("hour_letter") 

257 if hour_letter == "0" and hour == 0 and minute == 0: 

258 period = "01D" 

259 else: 

260 period = self.receiver_naming.period or self.site_naming.default_period 

261 

262 content = self.receiver_naming.content or self.site_naming.default_content 

263 agency = self.receiver_naming.agency or self.site_naming.agency 

264 file_type, compression = _detect_file_type(file_path) 

265 

266 conventional = CanVODFilename( 

267 site=self.site_naming.site_id, 

268 receiver_type=self._rx_type, 

269 receiver_number=self.receiver_naming.receiver_number, 

270 agency=agency, 

271 year=year, 

272 doy=doy, 

273 hour=hour, 

274 minute=minute, 

275 period=period, 

276 sampling=sampling, 

277 content=content, 

278 file_type=file_type, 

279 compression=compression, 

280 ) 

281 

282 return VirtualFile(physical_path=file_path, conventional_name=conventional)

296 def _discover_files(self) -> list[Path]: 

297 """Discover all data files according to directory layout.""" 

298 layout = self.receiver_naming.directory_layout 

299 

300 if layout == DirectoryLayout.FLAT: 

301 return self._glob_in_dir(self.receiver_base_dir) 

302 

303 if not self.receiver_base_dir.is_dir(): 

304 return [] 

305 

306 if layout == DirectoryLayout.YYDDD_SUBDIRS: 

307 dir_pattern = re.compile(r"^\d{5}$") 

308 else: 

309 dir_pattern = re.compile(r"^\d{7}$") 

310 

311 files: list[Path] = [] 

312 for subdir in sorted(self.receiver_base_dir.iterdir()): 

313 if subdir.is_dir() and dir_pattern.match(subdir.name): 

314 files.extend(self._glob_in_dir(subdir)) 

315 return files 

316 

317 def _glob_in_dir(self, directory: Path) -> list[Path]: 

318 """Glob for data files in a directory using the source pattern's globs.""" 

319 pattern_name = self.receiver_naming.source_pattern 

320 

321 if pattern_name == "auto": 

322 globs: set[str] = set() 

323 for name in auto_match_order(): 

324 globs.update(BUILTIN_PATTERNS[name].file_globs) 

325 else: 

326 globs = set(BUILTIN_PATTERNS[pattern_name].file_globs) 

327 

328 files: list[Path] = [] 

329 seen: set[Path] = set() 

330 for g in sorted(globs): 

331 for path in directory.glob(g): 

332 if path.is_file() and path not in seen: 

333 seen.add(path) 

334 files.append(path) 

335 return files
