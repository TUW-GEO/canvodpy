1"""Pipeline output reporters — plain (non-TTY) and Rich Live (TTY). 

2 

3Usage 

4----- 

5 rows = [(site_name, group_name, total_days), ...] # known upfront 

6 with make_reporter(rows) as r: 

7 r.set_current_site(site_name, start, end) 

8 r.print_header(site_name, start, end, config, args) 

9 r.on_day_start(date_key) 

10 r.on_datasets(datasets) 

11 r.on_vod_result(analysis, n_valid, n_total, dt) 

12 r.on_timing(dt_pipeline, dt_vod, dt_vod_store) 

13 r.advance(site_name, group_name) # called from an on_group_written callback 

14 r.on_done(total_days, total_vod, dt_total) 

15 

16Non-TTY (cron, CI, pipes): plain print() — identical to pre-dashboard output. 

17TTY: Rich Live display — one progress row per (site, receiver-group), known 

18upfront so a multi-site run shows every row from the start; per-day detail 

19scrolls above via live.console.print(). There is no aggregate "Overall" bar — 

20sites are processed sequentially, so the header panel's "current site" text 

21plus the per-row bars are the whole picture. 

22"""

41def day_count(start: str, end: str) -> int: 

42 """Estimate calendar days between two YYYYDOY strings (inclusive).""" 

43 parse = lambda s: datetime.strptime(s, "%Y%j") # noqa: E731 

44 return max(1, (parse(end) - parse(start)).days + 1)

52class PlainReporter: 

53 """Writes to stdout via print(). Behaviour identical to pre-dashboard output.""" 

54

62 def set_current_site(self, site: str, start: str, end: str) -> None: 

63 self._current_site = site 

64 print("=" * 72)

68 def print_header(self, site: str, start: str, end: str, config, args=None) -> None: 

69 proc = config.processing.params 

70 storage = config.processing.storage 

71 print(f" started {datetime.now():%Y-%m-%d %H:%M:%S}") 

72 print(f" ephemeris {proc.ephemeris_source}") 

73 print(f" keep_vars {proc.keep_gnss_observables}") 

74 if args is not None: 

75 print(f" days_per_batch {args.days_per_batch or proc.days_per_batch}") 

76 print(f" resource_mode {proc.resource_mode}") 

77 print(f" store_strategy {storage.gnss_store_strategy}") 

78 if args is not None: 

79 print(f" gnss_store {storage.gnss_store_name or 'rinex'}") 

80 print(f" vod_store {storage.vod_store_name or 'vod'}") 

81 print(f" vod {'skip' if args.no_vod else 'enabled'}") 

82 print()

84 def advance(self, site: str, group: str) -> None: 

85 pass # no bar to advance in plain mode

88 print(f"\n--- {self._current_site}/{date_key} ---") 

89 

90 def on_datasets(self, datasets: dict[str, xr.Dataset]) -> None:

92 e = ds.sizes.get("epoch", 0)

132class RichReporter: 

133 """Rich Live display: header panel + one progress row per (site, group). 

134 

135 Per-day detail lines are printed above the live region via 

136 live.console.print(), which Rich handles without flicker. 

137 """ 

138 

139 def __init__(self, rows: list[tuple[str, str, int]]) -> None: 

140 self._current_site = "" 

141 self._current_start = "" 

142 self._current_end = "" 

143 self._live: Live | None = None 

144 

145 from rich.console import Console 

146 from rich.progress import ( 

147 BarColumn, 

148 MofNCompleteColumn, 

149 Progress, 

150 SpinnerColumn, 

151 TaskID, # noqa: F401 

152 TextColumn, 

153 TimeElapsedColumn, 

154 TimeRemainingColumn, 

155 ) 

156 

157 self._console = Console() 

158 self._progress = Progress( 

159 SpinnerColumn(), 

160 TextColumn(" [bold]{task.description}[/bold]"), 

161 BarColumn( 

162 bar_width=None, complete_style="green3", finished_style="dim green" 

163 ), 

164 MofNCompleteColumn(), 

165 TextColumn("[dim]days[/dim]"), 

166 TimeElapsedColumn(), 

167 TextColumn("[dim]eta[/dim]"), 

168 TimeRemainingColumn(), 

169 console=self._console, 

170 ) 

171 self._task_ids: dict[tuple[str, str], TaskID] = { 

172 (site, group): self._progress.add_task(f"{site}/{group}", total=total) 

173 for site, group, total in rows 

174 } 

175 

176 def _render(self): 

177 try: 

178 from rich.group import Group # rich >= 12.0 

179 except ImportError: 

180 from rich.console import Group # rich < 12.0 

181 from rich.panel import Panel 

182 

183 site4 = self._current_site[:4].upper() if self._current_site else "----" 

184 content = f"─[◉]─ canvod · {site4} · {self._current_start}–{self._current_end}" 

185 return Group( 

186 Panel(content, border_style="dim green", padding=(0, 1)), 

187 self._progress, 

188 ) 

189 

190 @property 

191 def _live_obj(self) -> Live: 

192 assert self._live is not None, "must be used as a context manager" 

193 return self._live 

194 

195 def __enter__(self) -> RichReporter: 

196 from rich.live import Live 

197 

198 self._live = Live( 

199 self._render(), 

200 console=self._console, 

201 refresh_per_second=4, 

202 ) 

203 self._live.start(refresh=True) 

204 return self 

205 

206 def __exit__(self, *_) -> None: 

207 if self._live is not None: 

208 self._live.stop() 

209 

210 def log(self, msg: str) -> None: 

211 self._live_obj.console.print(msg) 

212 

213 def set_current_site(self, site: str, start: str, end: str) -> None: 

214 self._current_site = site 

215 self._current_start = start 

216 self._current_end = end 

217 self._live_obj.update(self._render()) 

218 self._live_obj.console.print(f"\n[bold green]═══ {site} ═══[/bold green]") 

219 

220 def print_header(self, site: str, start: str, end: str, config, args=None) -> None: 

221 proc = config.processing.params 

222 self._live_obj.console.print( 

223 f"[dim] ephemeris={proc.ephemeris_source}" 

224 f" resource_mode={proc.resource_mode}" 

225 f" strategy={config.processing.storage.gnss_store_strategy}[/dim]" 

226 ) 

227 

228 def advance(self, site: str, group: str) -> None: 

229 task_id = self._task_ids.get((site, group)) 

230 if task_id is not None: 

231 self._progress.advance(task_id) 

232 self._live_obj.update(self._render()) 

233 

234 def on_day_start(self, date_key: str) -> None: 

235 self._live_obj.console.print( 

236 f"\n[bold]─── {self._current_site}/{date_key}[/bold]" 

237 ) 

238 

239 def on_datasets(self, datasets: dict[str, xr.Dataset]) -> None: 

240 for group, ds in datasets.items(): 

241 e = ds.sizes.get("epoch", 0) 

242 s = ds.sizes.get("sid", 0) 

243 self._live_obj.console.print(f" [dim]{group}: {e}×{s}[/dim]") 

244 

245 def on_vod_result( 

246 self, analysis: str, n_valid: int, n_total: int, dt: float 

247 ) -> None: 

248 pct = 100 * n_valid / n_total if n_total else 0 

249 self._live_obj.console.print( 

250 f" [dim]VOD {analysis}: {n_valid}/{n_total} valid ({pct:.0f}%) {dt:.1f}s[/dim]" 

251 ) 

252 

253 def on_vod_failed(self, analysis: str, error: str) -> None: 

254 self._live_obj.console.print( 

255 f"[yellow] VOD {analysis}: FAILED — {error}[/yellow]" 

256 ) 

257 

258 def on_timing(self, dt_pipeline: float, dt_vod: float, dt_vod_store: float) -> None: 

259 self._live_obj.console.print( 

260 f" [dim]pipeline={dt_pipeline:.1f}s" 

261 f" vod={dt_vod:.1f}s" 

262 f" vod_store={dt_vod_store:.1f}s[/dim]" 

263 ) 

264 

265 def on_done(self, total_days: int, total_vod: int, dt_total: float) -> None: 

266 self._live_obj.console.print() 

267 self._live_obj.console.print( 

268 f"[bold green]Done {total_days} days " 

269 f"{total_vod} VOD analyses {dt_total:.0f}s total[/bold green]" 

270 )

278def make_reporter( 

279 rows: list[tuple[str, str, int]], 

280 *, 

281 tty: bool | None = None, 

282) -> PlainReporter | RichReporter: 

283 """Return a RichReporter when running in a TTY, PlainReporter otherwise. 

284 

285 Parameters 

286 ---------- 

287 rows : list[tuple[str, str, int]] 

288 One entry per (site_name, receiver_group, total_days) — every row 

289 known upfront so a multi-site run shows the full picture from the 

290 start. 

291 """ 

292 if tty is None: 

293 tty = sys.stdout.isatty() 

294 if tty: 

295 return RichReporter(rows) 

296 return PlainReporter(rows)
