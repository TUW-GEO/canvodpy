89def create_hemigrid( 

90 grid_type: Literal[ 

91 "equal_area", 

92 "equal_angle", 

93 "rectangular", 

94 "equirectangular", 

95 "HTM", 

96 "geodesic", 

97 "healpix", 

98 "fibonacci", 

99 ], 

100 angular_resolution: float = 10.0, 

101 **kwargs: Any, 

102) -> GridData: 

103 """Create hemisphere grid of specified type. 

104 

105 Factory function for creating various hemisphere grid types commonly 

106 used in GNSS analysis. 

107 

108 Parameters 

109 ---------- 

110 grid_type : str 

111 Type of grid to create: 

112 - 'equal_area': Regular lat/lon grid with equal solid angle cells 

113 - 'equal_angle': Regular angular spacing (not recommended) 

114 - 'rectangular' or 'equirectangular': Simple rectangular grid 

115 - 'HTM': Hierarchical Triangular Mesh 

116 - 'geodesic': Geodesic sphere subdivision (icosahedron-based) 

117 - 'healpix': HEALPix grid (requires healpy) 

118 - 'fibonacci': Fibonacci sphere (requires scipy) 

119 angular_resolution : float, default 10.0 

120 Angular resolution in degrees 

121 **kwargs 

122 Additional grid-specific parameters: 

123 - cutoff_theta : float - Maximum theta angle cutoff (degrees) 

124 - phi_rotation : float - Rotation angle (degrees) 

125 - subdivision_level : int - For geodesic/HTM grids 

126 - htm_level : int - For HTM grids specifically 

127 - nside : int - For HEALPix grids 

128 - n_points : int - For Fibonacci grids 

129 

130 Returns 

131 ------- 

132 GridData 

133 Complete hemisphere grid data structure 

134 

135 Examples 

136 -------- 

137 >>> # Equal area grid with 10° resolution 

138 >>> grid = create_hemigrid('equal_area', angular_resolution=10.0) 

139 >>> 

140 >>> # HTM grid with subdivision level 3 

141 >>> grid = create_hemigrid('HTM', angular_resolution=5.0, htm_level=3) 

142 >>> 

143 >>> # Geodesic grid 

144 >>> grid = create_hemigrid('geodesic', angular_resolution=5.0, subdivision_level=2) 

145 

146 Notes 

147 ----- 

148 Grid coordinates use navigation convention: 

149 - phi: azimuth angle, 0 to 2π (0 = North, π/2 = East, clockwise) 

150 - theta: polar angle from zenith, 0 to π/2 (0 = zenith, π/2 = horizon) 

151 

152 """ 

153 grid_type_lower = grid_type.lower() 

154 

155 if grid_type_lower == "equal_area": 

156 builder = EqualAreaBuilder( 

157 angular_resolution=angular_resolution, 

158 **kwargs, 

159 ) 

160 elif grid_type_lower == "equal_angle": 

161 builder = EqualAngleBuilder( 

162 angular_resolution=angular_resolution, 

163 **kwargs, 

164 ) 

165 elif grid_type_lower in ["rectangular", "equirectangular"]: 

166 builder = EquirectangularBuilder( 

167 angular_resolution=angular_resolution, 

168 **kwargs, 

169 ) 

170 elif grid_type_lower == "htm": 

171 builder = HTMBuilder( 

172 angular_resolution=angular_resolution, 

173 **kwargs, 

174 ) 

175 elif grid_type_lower == "geodesic": 

176 builder = GeodesicBuilder( 

177 angular_resolution=angular_resolution, 

178 **kwargs, 

179 ) 

180 elif grid_type_lower == "healpix": 

181 builder = HEALPixBuilder( 

182 angular_resolution=angular_resolution, 

183 **kwargs, 

184 ) 

185 elif grid_type_lower == "fibonacci": 

186 builder = FibonacciBuilder( 

187 angular_resolution=angular_resolution, 

188 **kwargs, 

189 ) 

190 else: 

191 raise ValueError( 

192 f"Unknown grid type: {grid_type}. " 

193 f"Available types: equal_area, equal_angle, rectangular, " 

194 f"equirectangular, HTM, geodesic, healpix, fibonacci" 

195 ) 

196 

197 return builder.build()
