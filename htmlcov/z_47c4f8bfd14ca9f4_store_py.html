1"""canvodpy store — inspect Icechunk stores from the terminal. 

2 

3Thin CLI wrappers around canvod-store's existing MyIcechunkStore methods 

4(print_tree, print_history, plot_commit_graph, print_ops_log,

31def _open_store(site_name: str, store_kind: str): 

32 """Resolve a site + store kind (gnss/vod) to an opened MyIcechunkStore. 

33 

34 Exits cleanly with an actionable message if the site is unknown or the 

35 store hasn't been created yet — matches the rest of the CLI's error 

36 style (config.py, doctor.py), not a raw traceback. 

37 """ 

38 from canvod.config import load_config 

39 from canvod.store import MyIcechunkStore 

40 

41 config = load_config() 

42 sites = config.sites.sites 

43 if site_name not in sites: 

44 console.print(f"[red]❌ Unknown site:[/red] {site_name}") 

45 known = ", ".join(sites) or "(none configured)" 

46 console.print(f" Known sites: {known}") 

47 raise typer.Exit(1) 

48 

49 storage = config.processing.storage 

50 if store_kind == "gnss": 

51 path = storage.get_gnss_store_path(site_name) 

52 store_type = "gnss_store" 

53 else: 

54 path = storage.get_vod_store_path(site_name) 

55 store_type = "vod_store" 

56 

57 if not path.exists(): 

58 console.print( 

59 f"[yellow]⊘ No {store_kind} store yet for '{site_name}'[/yellow] ({path})" 

60 ) 

61 raise typer.Exit(1) 

62 

63 return MyIcechunkStore(path, store_type=store_type)

143def _print_group_content(icestore, group_name: str, branch: str) -> None: 

144 """Terminal port of IcechunkStoreViewer's per-group HTML drill-down: 

145 plain-text xarray Dataset repr + metadata table, in a rich.Tree — 

146 same content, no browser/notebook required.""" 

147 tree = Tree(f"[bold]{group_name}[/bold] (branch: {branch})") 

148 

149 try: 

150 ds = icestore.read_group(group_name, branch=branch) 

151 tree.add(f"[bold]Dataset[/bold]\n{ds}") 

152 except Exception as e: 

153 tree.add(f"[red]Failed to load dataset: {e}[/red]") 

154 

155 try: 

156 with icestore.readonly_session(branch) as session: 

157 metadata_df = icestore.load_metadata(session.store, group_name) 

158 tree.add( 

159 f"[bold]Metadata table[/bold] ({len(metadata_df)} rows)\n{metadata_df}" 

160 ) 

161 except Exception: 

162 pass # not every group has a metadata table — fine, just skip it 

163 

164 console.print(tree)

202@store_app.command() 

203def maintain( 

204 site: Annotated[str, typer.Argument(help="Site name, as defined in sites.yaml")], 

205 store: Annotated[ 

206 str, typer.Option("--store", help="Which store: gnss or vod") 

207 ] = "gnss", 

208 expire_days: Annotated[ 

209 int, 

210 typer.Option( 

211 "--expire-days", 

212 help="Snapshot retention window in days (weeks-to-months, not hours)", 

213 ), 

214 ] = 90, 

215 gc: Annotated[ 

216 bool, typer.Option("--gc/--no-gc", help="Run garbage collection") 

217 ] = True, 

218 execute: Annotated[ 

219 bool, 

220 typer.Option( 

221 "--execute", 

222 help=( 

223 "Actually run expiration/GC. Without this flag, only a " 

224 "dry-run garbage-collect report is shown -- nothing is " 

225 "deleted or expired." 

226 ), 

227 ), 

228 ] = False, 

229) -> None: 

230 """Run store maintenance (expiration + garbage collection). 

231 

232 Defaults to a **dry run**: reports what garbage collection would delete 

233 and how many ancestry entries are older than the cutoff, without 

234 touching anything. Pass --execute to actually expire snapshots and 

235 (optionally) run garbage collection for real. 

236 

237 This is an administrative operation -- never safe to run casually 

238 alongside an active pipeline run on the same store. See 

239 dev/perf_degradation_findings_2026_07_15.md for the full rationale 

240 (expiration/GC concurrency caveats, why the default cadence is weeks- 

241 to-months, and the retention-scheme design this command supports). 

242 """ 

243 if store not in _STORE_KINDS: 

244 console.print( 

245 f"[red]❌ --store must be one of: {', '.join(_STORE_KINDS)}[/red]" 

246 ) 

247 raise typer.Exit(1) 

248 

249 from datetime import UTC, datetime, timedelta 

250 

251 icestore = _open_store(site, store) 

252 cutoff = datetime.now(UTC) - timedelta(days=expire_days) 

253 

254 if not execute: 

255 console.print( 

256 f"\n[bold]{site}[/bold] — {store} store — [yellow]DRY RUN[/yellow] " 

257 f"(cutoff: {cutoff.isoformat()})\n" 

258 ) 

259 

260 if gc: 

261 gc_result = icestore.garbage_collect(days=expire_days, dry_run=True) 

262 console.print("[bold]Would garbage-collect:[/bold]") 

263 for key in ( 

264 "bytes_deleted", 

265 "chunks_deleted", 

266 "manifests_deleted", 

267 "snapshots_deleted", 

268 "attributes_deleted", 

269 "transaction_logs_deleted", 

270 ): 

271 console.print(f" {key}: {gc_result[key]}") 

272 else: 

273 console.print("[dim]--no-gc: skipping garbage-collect dry run.[/dim]") 

274 

275 stale = sum( 

276 1 

277 for snap in icestore.repo.ancestry(branch="main") 

278 if snap.written_at < cutoff 

279 ) 

280 console.print(f"\n[bold]Ancestry entries older than cutoff:[/bold] {stale}") 

281 console.print( 

282 "\n[dim]Nothing was deleted or expired. Re-run with --execute " 

283 "to actually run maintenance (after confirming).[/dim]" 

284 ) 

285 return 

286 

287 console.print( 

288 f"\n[bold red]About to run maintenance on {site} ({store} store)[/bold red]" 

289 ) 

290 console.print(f" Cutoff: {cutoff.isoformat()} ({expire_days} days)") 

291 console.print(f" Garbage collection: {'yes' if gc else 'no'}") 

292 console.print( 

293 "\n[yellow]This is an administrative operation. Do not run it while " 

294 "a pipeline run is active against this store.[/yellow]" 

295 ) 

296 if not typer.confirm("Proceed?"): 

297 console.print("Aborted.") 

298 raise typer.Exit(0) 

299 

300 results = icestore.maintenance(expire_days=expire_days, run_gc=gc) 

301 console.print("\n[bold green]Maintenance complete:[/bold green]") 

302 console.print(f" Expired snapshots: {results['expired_snapshots']}") 

303 console.print(f" Deleted branches: {results['deleted_branches']}") 

304 if results["gc_summary"] is not None: 

305 console.print(f" GC summary: {results['gc_summary']}")

308@store_app.command("maintain-due") 

309def maintain_due( 

310 site: Annotated[ 

311 str | None, 

312 typer.Argument(help="Site name. Omit when using --all-sites."), 

313 ] = None, 

314 all_sites: Annotated[ 

315 bool, 

316 typer.Option("--all-sites", help="Run for every site in canvod-settings.yaml"), 

317 ] = False, 

318 store: Annotated[ 

319 str, 

320 typer.Option("--store", help="Which store(s): gnss, vod, or both"), 

321 ] = "both", 

322) -> None: 

323 """Non-interactive maintenance: runs only what's actually due, or nothing. 

324 

325 Cron/systemd-timer-safe counterpart to ``maintain`` -- never prompts, 

326 always exits 0 on a normal "nothing to do" outcome. `maintain` itself 

327 can't be scheduled unattended: its --execute path requires an 

328 interactive confirmation. This command instead reads 

329 ``processing.storage.maintenance`` from canvod-settings.yaml and does 

330 nothing unless all of the following hold: 

331 

332 \b 

333 - maintenance.enabled or maintenance.manifests_enabled is true (both 

334 default false -- inert until a deployment has validated `maintain 

335 --store ...` (dry run) against its real store first, see 

336 dev/perf_degradation_findings_2026_07_15.md) 

337 - no `canvodpy run` is currently active on this host 

338 

339 Expiration/GC are gated on maintenance.enabled specifically, and are 

340 due when the store's own ops log shows expiration/GC genuinely due 

341 per maintenance.expire_interval_days / .gc_delay_days. Manifest 

342 compaction (rewrite_manifests()) is gated independently by 

343 maintenance.manifests_enabled -- it's a different risk profile (an 

344 ordinary commit, no history rewrite or physical deletion) and has no 

345 ops-log entry of its own to check, so due-ness is measured directly 

346 from the store's current on-disk manifest count vs 

347 maintenance.manifest_count_threshold instead of a time interval. A 

348 deployment can enable either independently of the other. 

349 

350 While maintenance.dry_run_until_confirmed is true (the default), a due 

351 store is only ever reported, never actually touched -- flip it to 

352 false per-deployment once satisfied with the reports. 

353 """ 

354 if store == "both": 

355 store_kinds = _STORE_KINDS 

356 elif store in _STORE_KINDS: 

357 store_kinds = (store,) 

358 else: 

359 console.print( 

360 f"[red]❌ --store must be one of: both, {', '.join(_STORE_KINDS)}[/red]" 

361 ) 

362 raise typer.Exit(1) 

363 

364 if bool(site) == all_sites: 

365 console.print("[red]❌ Pass exactly one of: a site name, or --all-sites[/red]") 

366 raise typer.Exit(1) 

367 

368 from canvod.config import load_config 

369 from canvodpy.orchestrator.resources import is_pipeline_running 

370 

371 config = load_config() 

372 mcfg = config.processing.storage.maintenance 

373 

374 if not mcfg.enabled and not mcfg.manifests_enabled: 

375 console.print( 

376 "[dim]maintenance.enabled and maintenance.manifests_enabled are " 

377 "both false in canvod-settings.yaml -- nothing to do.[/dim]" 

378 ) 

379 return 

380 

381 if is_pipeline_running(): 

382 console.print( 

383 "[yellow]⊘ A canvodpy run is active on this host -- skipping " 

384 "(never run maintenance alongside active writes).[/yellow]" 

385 ) 

386 return 

387 

388 if all_sites: 

389 site_names = list(config.sites.sites) 

390 else: 

391 # The `bool(site) == all_sites` guard above already ensures site is 

392 # truthy here (exactly one of site/--all-sites is set) -- assert 

393 # narrows the type for ty rather than leaving it `str | None`. 

394 assert site is not None 

395 if site not in config.sites.sites: 

396 console.print(f"[red]❌ Unknown site:[/red] {site}") 

397 raise typer.Exit(1) 

398 site_names = [site] 

399 

400 for site_name in site_names: 

401 for kind in store_kinds: 

402 storage = config.processing.storage 

403 path = ( 

404 storage.get_gnss_store_path(site_name) 

405 if kind == "gnss" 

406 else storage.get_vod_store_path(site_name) 

407 ) 

408 if not path.exists(): 

409 console.print( 

410 f"[dim]{site_name} ({kind}): no store yet, skipping.[/dim]" 

411 ) 

412 continue 

413 

414 from canvod.store import MyIcechunkStore 

415 

416 icestore = MyIcechunkStore( 

417 path, store_type="gnss_store" if kind == "gnss" else "vod_store" 

418 ) 

419 due = icestore.maintenance_due( 

420 expire_interval_days=mcfg.expire_interval_days, 

421 gc_delay_days=mcfg.gc_delay_days, 

422 manifest_count_threshold=( 

423 mcfg.manifest_count_threshold if mcfg.manifests_enabled else None 

424 ), 

425 ) 

426 if not mcfg.enabled: 

427 # expire/GC stay gated on `enabled` specifically -- 

428 # `manifests_enabled` only ever authorizes compaction. 

429 due["expire_due"] = False 

430 due["gc_due"] = False 

431 

432 if not due["expire_due"] and not due["gc_due"] and not due["manifests_due"]: 

433 console.print( 

434 f"[dim]{site_name} ({kind}): not due " 

435 f"(last expire: {due['last_expire']}, " 

436 f"last gc: {due['last_gc']}, " 

437 f"manifests: {due['manifest_count']}).[/dim]" 

438 ) 

439 continue 

440 

441 console.print( 

442 f"\n[bold]{site_name}[/bold] ({kind}) -- due: " 

443 f"expire={due['expire_due']} gc={due['gc_due']} " 

444 f"manifests={due['manifests_due']}" 

445 ) 

446 

447 if mcfg.dry_run_until_confirmed: 

448 console.print( 

449 "[yellow]DRY RUN[/yellow] (maintenance.dry_run_until_confirmed=true)" 

450 ) 

451 if due["gc_due"]: 

452 gc_result = icestore.garbage_collect( 

453 days=mcfg.retention_days, dry_run=True 

454 ) 

455 console.print(f" Would garbage-collect: {gc_result}") 

456 if due["expire_due"]: 

457 from datetime import UTC, datetime, timedelta 

458 

459 cutoff = datetime.now(UTC) - timedelta(days=mcfg.retention_days) 

460 stale = sum( 

461 1 

462 for snap in icestore.repo.ancestry(branch="main") 

463 if snap.written_at < cutoff 

464 ) 

465 console.print( 

466 f" Would expire: {stale} ancestry entries older than cutoff" 

467 ) 

468 if due["manifests_due"]: 

469 # rewrite_manifests() has no dry-run mode -- report the 

470 # fragmentation count that triggered due-ness instead. 

471 console.print( 

472 f" Would compact manifests: {due['manifest_count']} " 

473 f">= threshold {mcfg.manifest_count_threshold}" 

474 ) 

475 continue 

476 

477 if due["expire_due"]: 

478 expired = icestore.expire_old_snapshots(days=mcfg.retention_days) 

479 console.print(f" Expired: {len(expired)} snapshots") 

480 if due["gc_due"]: 

481 gc_result = icestore.garbage_collect( 

482 days=mcfg.retention_days, dry_run=False 

483 ) 

484 console.print(f" GC: {gc_result}") 

485 if due["manifests_due"]: 

486 snapshot_id = icestore.compact_manifests(branch="main") 

487 console.print(f" Compacted manifests: snapshot {snapshot_id}")
