60 def __call__(self, ds: xr.Dataset) -> tuple[xr.Dataset, OpResult]: 

61 t0 = time.perf_counter() 

62 params = self.settings 

63 input_shape = {str(k): int(v) for k, v in dict(ds.sizes).items()} 

64 

65 # Prerequisite check: phi/theta as data variables (as the runs write 

66 # them) or as coordinates 

67 has_phi = "phi" in ds.variables and set(ds["phi"].dims) == {"epoch", "sid"} 

68 has_theta = "theta" in ds.variables and set(ds["theta"].dims) == { 

69 "epoch", 

70 "sid", 

71 } 

72 

73 if not (has_phi and has_theta): 

74 logger.warning( 

75 "Grid assignment skipped: dataset has no phi/theta (epoch, sid) variables" 

76 ) 

77 result = OpResult( 

78 op_name=self.name, 

79 parameters=params, 

80 input_shape=input_shape, 

81 output_shape=input_shape, 

82 duration_seconds=time.perf_counter() - t0, 

83 notes="skipped: missing phi/theta", 

84 result={"assigned": False, "reason": "no phi/theta (epoch, sid)"}, 

85 ) 

86 return ds, result

89 grid_name = f"{self._grid_type}_{self._angular_resolution}deg" 

90 

91 from canvod.grids.operations import add_cell_ids_to_ds_fast

95

138def assign_grid_cells_in_store( 

139 store: MyIcechunkStore, 

140 group_name: str, 

141 grid_type: str, 

142 angular_resolution: float, 

143 branch: str = "main", 

144) -> str: 

145 """Assign the observations already in a store group to grid cells. 

146 

147 The same operation as ``processing.preprocessing.grid_assignment`` in a 

148 run, applied afterwards to all data of ``group_name`` (a GNSS receiver 

149 group or a VOD analysis group). Only the new ``cell_id_<grid>`` variable 

150 is written; the data, the log book and the other subgroups stay as they 

151 are (``MyIcechunkStore.add_group_variables``). 

152 

153 Later writes into the group must carry the cells too: a run is accepted 

154 only with ``processing.preprocessing.grid_assignment`` set to the same 

155 ``grid_type`` and ``angular_resolution``. A VOD analysis group accepts 

156 only VOD computed from GNSS data with the same grid assignment, so 

157 assign the cells to the receiver groups as well. 

158 

159 Parameters 

160 ---------- 

161 store : MyIcechunkStore 

162 GNSS or VOD store. 

163 group_name : str 

164 Group whose observations to assign. 

165 grid_type : str 

166 Grid builder name (e.g. ``"equal_area"``). 

167 angular_resolution : float 

168 Grid resolution in degrees. 

169 branch : str, default "main" 

170 Branch to change. 

171 

172 Returns 

173 ------- 

174 str 

175 The new tip of ``branch``. 

176 

177 Raises 

178 ------ 

179 ValueError 

180 If the group has no ``phi`` / ``theta`` on ``(epoch, sid)``, already 

181 has these cells, or a temporary branch is left from an earlier call. 

182 """ 

183 op = GridAssignment(grid_type=grid_type, angular_resolution=angular_resolution) 

184 ds = store.read_group(group_name, branch=branch) 

185 out, result = op(ds) 

186 if not result.result.get("assigned"): 

187 raise ValueError( 

188 f"Group '{group_name}' has no phi/theta (epoch, sid) variables to " 

189 "assign to grid cells" 

190 ) 

191 variable = result.result["variable"] 

192 return store.add_group_variables( 

193 group_name, 

194 out[[variable]], 

195 step=result.step(), 

196 software=software_versions(GridAssignment.packages), 

197 branch=branch, 

198 )
