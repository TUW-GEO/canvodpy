1"""RINEX 3.04/3.05 navigation files: the broadcast ephemeris records. 

2 

3A navigation file holds one record per satellite and reference time: the 

4orbit parameters the satellite broadcasts. :func:`read_rinex_nav` reads the 

5records of one or more files into one dataset per system, each with a 

6``record`` dimension. 

7 

8Read so far: GPS (RINEX 3.05, Table A6), Galileo (Table A8), GLONASS 

9(Table A10), QZSS (Table A12), BeiDou (Table A14) and SBAS (Table A16). 

10Records of the other systems are skipped. Times of clock are converted to 

11GPS time, like the observation epochs; times of ephemeris and weeks stay 

12as written, in the system's time. 

13 

14The record layout is fixed-width: a record starts with the satellite 

15(``A1,I2.2``) and its epoch on the first line, followed by lines of four 

16``D19.12`` fields after four blanks (RINEX 3.05, Section 6 and Table A6). 

17"""

226def _header(lines: list[str], path: Path) -> int: 

227 """Check the header of a navigation file; index of its first record line.""" 

228 first = lines[0] if lines else "" 

229 version = first[:9].strip() 

230 if "RINEX VERSION / TYPE" not in first or first[20:21] != "N": 

231 raise ValueError(f"{path} is not a RINEX navigation file") 

232 if version not in SUPPORTED_VERSIONS: 

233 raise ValueError( 

234 f"{path} is RINEX {version}; navigation files are read in RINEX " 

235 f"{' and '.join(SUPPORTED_VERSIONS)}" 

236 ) 

237 for i, line in enumerate(lines): 

238 if "END OF HEADER" in line[60:]: 

239 return i + 1 

240 raise ValueError(f"{path} has no END OF HEADER line")

243def _records(lines: list[str], start: int) -> Iterable[list[str]]: 

244 """The lines of each record: a record line starts with its satellite.""" 

245 record: list[str] = [] 

246 for line in lines[start:]: 

247 if not line.strip(): 

248 continue 

249 if line[0] != " " and record: 

250 yield record 

251 record = [] 

252 record.append(line) 

253 if record: 

254 yield record

257def _fields(record: list[str]) -> list[str]: 

258 """The numeric fields of a record, as text: 3 on the first line, 4 after.""" 

259 first = record[0] 

260 fields = [ 

261 first[23 + _FIELD_WIDTH * k : 23 + _FIELD_WIDTH * (k + 1)] for k in range(3) 

262 ] 

263 for line in record[1:]: 

264 fields += [ 

265 line[4 + _FIELD_WIDTH * k : 4 + _FIELD_WIDTH * (k + 1)] for k in range(4) 

266 ] 

267 return fields

270def _utc_minus_gps(toc: np.ndarray) -> np.ndarray: 

271 """UTC minus GPS time at UTC times: minus the leap seconds of each.""" 

272 times = np.unique(toc) 

273 leap = [gps_minus_utc(t.astype("datetime64[us]").item()) for t in times] 

274 return -np.array(leap, dtype="timedelta64[s]")[np.searchsorted(times, toc)]

277def _system_records( 

278 system: str, svs: list[str], epochs: list[str], texts: list[list[str]] 

279) -> xr.Dataset: 

280 """The records of one system as a dataset along ``record``.""" 

281 fields = FIELDS[system] 

282 values = np.char.replace(np.array(texts, dtype=str), "D", "E") 

283 values = np.where(np.char.strip(values) == "", "nan", values).astype(np.float64) 

284 toc = np.array(epochs, dtype="datetime64[ns]") 

285 if system == "R": 

286 toc = toc - _utc_minus_gps(toc) 

287 else: 

288 toc = toc - _TOC_MINUS_GPS.get(system, np.timedelta64(0, "s")) 

289 sv = np.array(svs, dtype=str) 

290 

291 keys = [sv, toc] + [ 

292 np.nan_to_num(values[:, fields.index(name)], nan=-1.0) 

293 for name in _RECORD_KEYS[system] 

294 ] 

295 _, first = np.unique(np.rec.fromarrays(keys), return_index=True) 

296 order = first[np.lexsort((toc[first], sv[first]))] 

297 

298 ds = xr.Dataset( 

299 { 

300 name: ("record", values[order, i]) 

301 for i, name in enumerate(fields) 

302 if name is not None 

303 }, 

304 coords={"sv": ("record", sv[order]), "toc": ("record", toc[order])}, 

305 ) 

306 ds["toc"].attrs["time_system"] = "GPS" 

307 ds.attrs["source"] = f"RINEX navigation files ({_SOURCES[system]})" 

308 return ds

311def read_rinex_nav(paths: Iterable[Path]) -> dict[str, xr.Dataset]: 

312 """Read the broadcast ephemeris records of RINEX navigation files. 

313 

314 Parameters 

315 ---------- 

316 paths : Iterable[Path] 

317 RINEX 3.04/3.05 navigation files, for example the files of one 

318 receiver-day. A record in several files is kept once. 

319 

320 Returns 

321 ------- 

322 dict[str, xr.Dataset] 

323 One dataset per system letter found (``"G"``, ``"E"``, ``"R"``, 

324 ``"J"``, ``"C"``, ``"S"``), with one entry per record along 

325 ``record``, sorted by satellite and time of clock: coordinates 

326 ``sv`` (e.g. ``"G01"``) and ``toc`` (time of clock, GPS time), and 

327 one variable per field of the system's table in :data:`FIELDS`. 

328 

329 Raises 

330 ------ 

331 ValueError 

332 If no files are given, a file is not a RINEX 3.04/3.05 navigation 

333 file, or the files hold no record of a system read. 

334 """ 

335 paths = [Path(path) for path in paths] 

336 if not paths: 

337 raise ValueError("no navigation files given") 

338 found: dict[str, tuple[list[str], list[str], list[list[str]]]] = {} 

339 for path in paths: 

340 lines = path.read_text().splitlines() 

341 for record in _records(lines, _header(lines, path)): 

342 system = record[0][0] 

343 fields = FIELDS.get(system) 

344 if fields is None: 

345 continue 

346 # QZSS writes the number as I2, the others as I2.2 (Table A12) 

347 sv = f"{system}{int(record[0][1:3]):02d}" 

348 year, month, day, hour, minute, second = map(int, record[0][4:23].split()) 

349 svs, epochs, texts = found.setdefault(system, ([], [], [])) 

350 svs.append(sv) 

351 epochs.append( 

352 f"{year:04d}-{month:02d}-{day:02d}T{hour:02d}:{minute:02d}:{second:02d}" 

353 ) 

354 texts.append((_fields(record) + [""] * len(fields))[: len(fields)]) 

355 if not found: 

356 raise ValueError( 

357 f"{', '.join(str(path) for path in paths)} hold no record of " 

358 f"{', '.join(sorted(FIELDS))}; navigation records of the other " 

359 "systems are not read yet" 

360 ) 

361 return { 

362 system: _system_records(system, *columns) 

363 for system, columns in sorted(found.items()) 

364 }
