6def phi_bbox(phi_values: np.ndarray) -> tuple[float, float]: 

7 """Angular bounding box of a cluster of phi values in [0, 2*pi). 

8 

9 Naive ``(min(phi), max(phi))`` reports a near-full-circle span for a 

10 cell whose vertices straddle the 0/2*pi seam (e.g. vertices at 0.01 and 

11 6.27 rad give ``[0.01, 6.27]`` instead of the true ~0.02 rad width). 

12 Assumes ``phi_values`` are a single angularly-localized cluster (true 

13 for one cell's own vertices) -- computes the bbox both as-is and in a 

14 frame shifted by pi, and keeps whichever framing gives the narrower 

15 span. 

16 

17 Parameters 

18 ---------- 

19 phi_values : np.ndarray 

20 Azimuthal angles of a cell's vertices, radians (any range; wrapped 

21 into [0, 2*pi) internally). 

22 

23 Returns 

24 ------- 

25 phi_min, phi_max : float 

26 ``phi_min`` is in [0, 2*pi). ``phi_max`` may exceed 2*pi when the 

27 cell wraps the seam -- callers should use ``phi_max - phi_min`` 

28 directly for angular width rather than assuming both bounds lie in 

29 [0, 2*pi); polar axes wrap angles > 2*pi visually. 

30 

31 """ 

32 phi_values = np.asarray(phi_values, dtype=np.float64) % (2 * np.pi) 

33 lo, hi = float(phi_values.min()), float(phi_values.max()) 

34 span = hi - lo 

35 

36 shifted = (phi_values + np.pi) % (2 * np.pi) 

37 lo_s, hi_s = float(shifted.min()), float(shifted.max()) 

38 span_s = hi_s - lo_s 

39 

40 if span_s < span: 

41 lo, hi = lo_s - np.pi, hi_s - np.pi 

42 if lo < 0: 

43 lo += 2 * np.pi 

44 hi += 2 * np.pi 

45 

46 return lo, hi
