1"""Logging configuration for canvodpy with dual-output system. 

2 

3This module configures structured logging with multiple outputs: 

4- Agent-diagnostic JSON logs (machine/agent.json) -- a curated causal 

5 narrative (run_id, stage transitions, degraded-path events, crashes) 

6 meant to let a human or an AI agent reconstruct what happened on a 

7 remote, unattended run from the log file alone. 

8- Human-readable text logs for interactive debugging 

9- Component-specific logs for isolation 

10- Log rotation for manageable file sizes 

11 

12Every log record automatically carries the current ``run_id`` (see 

13``run_context.py``), and any exception that would otherwise escape the 

14process uncaught is captured here via ``sys.excepthook`` before the 

15process dies -- since on a remote run, the log is the only evidence that 

16will ever exist. 

17"""

35def _process_log_suffix() -> str: 

36 """Return a filename suffix identifying the current OS process. 

37 

38 Empty in the main process (unchanged filenames, backwards compatible). 

39 Non-empty in any other process (e.g. a loky/ProcessPoolExecutor worker), 

40 since ``RotatingFileHandler``/``TimedRotatingFileHandler`` are only safe 

41 within a single process — multiple processes rotating the same physical 

42 file race on ``os.rename()``. Giving each process its own file set 

43 removes the shared path entirely instead of trying to coordinate rotation 

44 across processes. 

45 """ 

46 if multiprocessing.current_process().name == "MainProcess": 

47 return "" 

48 return f".{os.getpid()}"

77def _create_human_renderer() -> Callable[..., Any]: 

78 """Create human-readable formatter function. 

79 

80 Returns 

81 ------- 

82 Callable 

83 Processor function for human-readable logs. 

84 """ 

85 

86 def human_format(logger, name, event_dict): 

87 """Format log entry in traditional log format.""" 

88 timestamp = event_dict.get("timestamp", "") 

89 level = event_dict.get("level", "INFO").upper() 

90 event = event_dict.get("event", "") 

91 

92 # Extract context fields 

93 context_parts = [] 

94 for key, value in sorted(event_dict.items()): 

95 # Skip internal/standard fields 

96 if key in ("timestamp", "level", "event", "logger", "exc_info"): 

97 continue 

98 if key.startswith("_"): 

99 continue 

100 context_parts.append(f"{key}={value}") 

101 

102 context = f" [{', '.join(context_parts)}]" if context_parts else "" 

103 

104 return f"{timestamp} {level:8s} {event}{context}" 

105 

106 return human_format

109def _create_error_renderer() -> Callable[..., Any]: 

110 """Create detailed error formatter with stack traces. 

111 

112 Returns 

113 ------- 

114 Callable 

115 Processor function for error logs with full context. 

116 """ 

117 

118 def error_format(logger, name, event_dict): 

119 """Format error log entry with full details.""" 

120 timestamp = event_dict.get("timestamp", "") 

121 level = event_dict.get("level", "INFO").upper() 

122 event = event_dict.get("event", "") 

123 logger_name = event_dict.get("logger", "") 

124 

125 # Build header 

126 lines = [ 

127 f"{'=' * 80}", 

128 f"{timestamp} {level} [{logger_name}]", 

129 f"Event: {event}", 

130 ] 

131 

132 # Add context fields 

133 context_items = [] 

134 for key, value in sorted(event_dict.items()): 

135 if key in ( 

136 "timestamp", 

137 "level", 

138 "event", 

139 "logger", 

140 "exc_info", 

141 "exception", 

142 ): 

143 continue 

144 if key.startswith("_"): 

145 continue 

146 context_items.append(f" {key}: {value}") 

147 

148 if context_items: 

149 lines.append("Context:") 

150 lines.extend(context_items) 

151 

152 # Add exception if present 

153 exc_info = event_dict.get("exc_info") 

154 exception = event_dict.get("exception") 

155 if exception: 

156 lines.append("Exception:") 

157 lines.append(exception) 

158 elif exc_info: 

159 lines.append(f"Exception Info: {exc_info}") 

160 

161 lines.append("") # Blank line between errors 

162 return "\n".join(lines) 

163 

164 return error_format

175class PerformanceFilter(logging.Filter): 

176 """Filter that only passes performance-timing logs. 

177 

178 Matches the canonical event emitted by ``stage_timer()`` 

179 (``event="stage_timing"``), plus any event that carries a 

180 ``duration_seconds`` field under its own specific event name -- many 

181 call sites (e.g. ``ephemeris_interpolation_complete``, 

182 ``aux_pipeline_create_standard_complete``) are deliberately detailed, 

183 named events that already report timing alongside other business 

184 context (shapes, dims, counts); they're not a "ragged" convention that 

185 needs consolidating into a generic event name, just correctly-named 

186 already. Only the genuinely inconsistent field names 

187 (``processing_time_min``, ``hampel_processing_time_s``, etc.) needed 

188 fixing, not these. 

189 """ 

190 

191 def filter(self, record): 

192 """Return True for stage_timing events or any duration_seconds field.""" 

193 if hasattr(record, "msg") and isinstance(record.msg, dict): 

194 msg = record.msg 

195 return msg.get("event") == "stage_timing" or "duration_seconds" in msg 

196 return False

199class ComponentFilter(logging.Filter): 

200 """Filter that only passes logs from specific component.""" 

201 

202 def __init__(self, component_name: str): 

203 """Initialize filter for specific component. 

204 

205 Parameters 

206 ---------- 

207 component_name : str 

208 Name of component to filter (e.g., "processor", "auxiliary") 

209 """ 

210 super().__init__() 

211 self.component_name = component_name 

212 

213 def filter(self, record): 

214 """Return True only for logs from this component.""" 

215 # Check logger name 

216 if self.component_name in record.name: 

217 return True 

218 

219 # Check if log message dict has component field 

220 if ( 

221 hasattr(record, "msg") 

222 and isinstance(record.msg, dict) 

223 and record.msg.get("component") == self.component_name 

224 ): 

225 return True 

226 

227 # Check if record has component attribute (set by ProcessorFormatter) 

228 return hasattr(record, "component") and record.component == self.component_name

243def _install_excepthook(logger: structlog.BoundLogger) -> None: 

244 """Install a process-wide safety net for uncaught exceptions. 

245 

246 On a remote, unattended run there is no re-run and no debugger -- 

247 the log file is the only forensic evidence that will ever exist. Any 

248 exception that reaches here would otherwise propagate to Python's 

249 default handler and be lost if stderr isn't captured, so this logs a 

250 structured ``uncaught_exception`` event (with the current run_id and 

251 full traceback) before chaining to the default hook, which still runs 

252 so console behavior for attended sessions is unchanged. 

253 

254 Idempotent: ``configure_logging()`` runs once per process in normal 

255 operation (module-level ``LOGGER = configure_logging()``), but can run 

256 more than once (tests, or a hypothetical re-configuration). Without 

257 this guard, each call would wrap the *previous* call's hook, growing an 

258 ever-longer chain that re-logs the same exception once per wrap. The 

259 marker lives on ``sys.excepthook`` itself (not separate module state), 

260 so if something resets ``sys.excepthook`` externally between calls 

261 (e.g. a test fixture), the next call correctly reinstalls rather than 

262 silently no-op'ing against a hook that's no longer actually wired up. 

263 """ 

264 if getattr(sys.excepthook, "_is_canvodpy_excepthook", False): 

265 sys.excepthook._logger_holder[0] = logger # ty: ignore[unresolved-attribute] 

266 return 

267 

268 default_hook = sys.excepthook 

269 logger_holder = [logger] 

270 

271 def _excepthook(exc_type, exc_value, exc_tb): 

272 if not issubclass(exc_type, KeyboardInterrupt): 

273 try: 

274 logger_holder[0].error( 

275 "uncaught_exception", 

276 exc_type=exc_type.__name__, 

277 exc_message=str(exc_value), 

278 traceback="".join( 

279 traceback.format_exception(exc_type, exc_value, exc_tb) 

280 ), 

281 ) 

282 except Exception: 

283 # The crash handler must never itself crash the crash report. 

284 pass 

285 default_hook(exc_type, exc_value, exc_tb) 

286 

287 _excepthook._is_canvodpy_excepthook = True # ty: ignore[unresolved-attribute] 

288 _excepthook._logger_holder = logger_holder # ty: ignore[unresolved-attribute] 

289 sys.excepthook = _excepthook

292def configure_logging(logfile: Path | None = None) -> structlog.BoundLogger: 

293 """Configure structlog with dual-output system. 

294 

295 Creates organized logging structure: 

296 - machine/agent.json - Always-on agent-diagnostic causal narrative 

297 - human/main.log - Traditional readable logs 

298 - Console output for warnings/errors 

299 

300 All file handlers include log rotation. 

301 

302 Parameters 

303 ---------- 

304 logfile : Path, optional 

305 Legacy parameter for backwards compatibility. Now used to determine 

306 base log directory. Defaults to LOG_FILE. 

307 

308 Returns 

309 ------- 

310 structlog.BoundLogger 

311 Configured structlog logger. 

312 """ 

313 if logfile is None: 

314 try: 

315 logfile = load_config().processing.logging.get_log_file() 

316 except Exception: 

317 # Fallback when config system isn't available (e.g. CI, tests) 

318 logfile = Path.cwd() / ".logs" / "canvodpy.log" 

319 

320 # Set up directory structure 

321 base_dir = logfile.parent 

322 log_dirs = _setup_log_directories(base_dir) 

323 

324 # Distinguishes worker-process log files from the main process's, so 

325 # concurrent processes never share a physical path (see _process_log_suffix). 

326 suffix = _process_log_suffix() 

327 

328 timestamper = structlog.processors.TimeStamper(fmt="iso", utc=True) 

329 

330 shared_processors = [ 

331 timestamper, 

332 structlog.processors.add_log_level, 

333 _add_run_id, 

334 structlog.processors.StackInfoRenderer(), 

335 structlog.processors.UnicodeDecoder(), 

336 structlog.processors.format_exc_info, 

337 ] 

338 

339 # Renderers 

340 console_renderer = structlog.dev.ConsoleRenderer() 

341 json_renderer = structlog.processors.JSONRenderer() 

342 

343 # Reset root logger 

344 logging.basicConfig(level=logging.DEBUG, format="%(message)s") 

345 root_logger = logging.getLogger() 

346 root_logger.handlers.clear() 

347 root_logger.setLevel(logging.DEBUG) # Ensure root accepts all levels 

348 

349 # ======================================================================== 

350 # HANDLER 1: Console (stdout) - Warnings and errors only 

351 # ======================================================================== 

352 console_handler = logging.StreamHandler(sys.stdout) 

353 console_handler.setLevel(logging.WARNING) 

354 console_formatter = structlog.stdlib.ProcessorFormatter( 

355 foreign_pre_chain=shared_processors, 

356 processors=[ 

357 structlog.stdlib.ProcessorFormatter.remove_processors_meta, 

358 console_renderer, 

359 ], 

360 ) 

361 console_handler.setFormatter(console_formatter) 

362 

363 # ======================================================================== 

364 # HANDLER 2: Agent-diagnostic JSON - always-on DEBUG causal narrative 

365 # ======================================================================== 

366 # Deliberately always-on (never gated behind a debug flag): on a remote 

367 # unattended run there is no way to retroactively enable debug logging 

368 # after a failure, so this must already be capturing everything. 

369 agent_log = log_dirs["machine"] / f"agent{suffix}.json" 

370 agent_handler = RotatingFileHandler( 

371 agent_log, 

372 maxBytes=100 * 1024 * 1024, # 100MB max per file 

373 # backupCount=10 (1GB) only reached ~1h back on a 15h overnight run 

374 # (~2.2GB/day observed, 2026-07-15) -- found by hitting it directly, 

375 # while investigating that same run's write-side degradation: 

376 # agent.json.10 (the oldest kept backup) started at 23:14 UTC, but 

377 # the run began at 22:15, so the first hour of debug detail for that 

378 # run was already gone. 150 (~15GB, ~7 days at this rate) is a 

379 # stopgap sized for "don't lose crash forensics within about a 

380 # week," not a durable fix -- same caveat as performance.json's 

381 # backupCount bump (see dev/todo_later.md §37): the real answer is 

382 # exporting before rotation can touch it. 

383 backupCount=150, 

384 encoding="utf-8", 

385 ) 

386 agent_handler.setLevel(logging.DEBUG) # Capture everything 

387 agent_formatter = structlog.stdlib.ProcessorFormatter( 

388 foreign_pre_chain=shared_processors, 

389 processors=[ 

390 structlog.stdlib.ProcessorFormatter.remove_processors_meta, 

391 json_renderer, 

392 ], 

393 ) 

394 agent_handler.setFormatter(agent_formatter) 

395 

396 # ======================================================================== 

397 # HANDLER 3: Human-readable - Traditional logs with rotation 

398 # ======================================================================== 

399 human_log = log_dirs["human"] / f"main{suffix}.log" 

400 human_handler = TimedRotatingFileHandler( 

401 human_log, 

402 when="midnight", # Rotate at midnight 

403 interval=1, # Daily rotation 

404 backupCount=30, # Keep 30 days 

405 encoding="utf-8", 

406 ) 

407 human_handler.setLevel(logging.INFO) 

408 human_formatter = structlog.stdlib.ProcessorFormatter( 

409 foreign_pre_chain=shared_processors, 

410 processors=[ 

411 structlog.stdlib.ProcessorFormatter.remove_processors_meta, 

412 _create_human_renderer(), 

413 ], 

414 ) 

415 human_handler.setFormatter(human_formatter) 

416 

417 # ======================================================================== 

418 # HANDLER 4: Errors Only - Detailed error logs with stack traces 

419 # ======================================================================== 

420 error_log = log_dirs["human"] / f"errors{suffix}.log" 

421 error_handler = RotatingFileHandler( 

422 error_log, 

423 maxBytes=50 * 1024 * 1024, # 50MB max per file 

424 backupCount=5, # Keep 5 rotated files 

425 encoding="utf-8", 

426 ) 

427 error_handler.setLevel(logging.ERROR) 

428 error_handler.addFilter(ErrorFilter()) 

429 error_formatter = structlog.stdlib.ProcessorFormatter( 

430 foreign_pre_chain=shared_processors, 

431 processors=[ 

432 structlog.stdlib.ProcessorFormatter.remove_processors_meta, 

433 _create_error_renderer(), 

434 ], 

435 ) 

436 error_handler.setFormatter(error_formatter) 

437 

438 # ======================================================================== 

439 # HANDLER 5: Performance Metrics - Timing and throughput data 

440 # ======================================================================== 

441 perf_log = log_dirs["machine"] / f"performance{suffix}.json" 

442 perf_handler = RotatingFileHandler( 

443 perf_log, 

444 maxBytes=50 * 1024 * 1024, # 50MB max per file 

445 # backupCount=10 (500MB) silently deleted history after ~5 days on a 

446 # real overnight backfill (~96MB/day observed, 2026-07-15) -- too 

447 # short for the "runs may take days, or are triggered daily and run 

448 # forever" case. 60 (~3GB/~31 days) is a stopgap, not a fix: the 

449 # durable answer is exporting machine/performance*.json into a 

450 # persistent store before rotation can touch it (dev/todo_later.md 

451 # §34 Tier 1, not yet built). 

452 backupCount=60, 

453 encoding="utf-8", 

454 ) 

455 perf_handler.setLevel(logging.DEBUG) 

456 perf_handler.addFilter(PerformanceFilter()) 

457 perf_formatter = structlog.stdlib.ProcessorFormatter( 

458 foreign_pre_chain=shared_processors, 

459 processors=[ 

460 structlog.stdlib.ProcessorFormatter.remove_processors_meta, 

461 json_renderer, 

462 ], 

463 ) 

464 perf_handler.setFormatter(perf_formatter) 

465 

466 # ======================================================================== 

467 # HANDLER 6-8: Component-specific logs 

468 # ======================================================================== 

469 component_handlers = [] 

470 for component_name in ["processor", "auxiliary", "icechunk"]: 

471 comp_log = log_dirs["component"] / f"{component_name}{suffix}.log" 

472 comp_handler = TimedRotatingFileHandler( 

473 comp_log, 

474 when="midnight", 

475 interval=1, 

476 backupCount=7, # Keep 1 week 

477 encoding="utf-8", 

478 ) 

479 comp_handler.setLevel(logging.DEBUG) 

480 comp_handler.addFilter(ComponentFilter(component_name)) 

481 comp_formatter = structlog.stdlib.ProcessorFormatter( 

482 foreign_pre_chain=shared_processors, 

483 processors=[ 

484 structlog.stdlib.ProcessorFormatter.remove_processors_meta, 

485 _create_human_renderer(), 

486 ], 

487 ) 

488 comp_handler.setFormatter(comp_formatter) 

489 component_handlers.append(comp_handler) 

490 

491 # Register all handlers 

492 root_logger.addHandler(console_handler) 

493 root_logger.addHandler(agent_handler) 

494 root_logger.addHandler(human_handler) 

495 root_logger.addHandler(error_handler) 

496 root_logger.addHandler(perf_handler) 

497 for handler in component_handlers: 

498 root_logger.addHandler(handler) 

499 

500 # Structlog config 

501 structlog.configure( 

502 processors=[ 

503 *shared_processors, 

504 structlog.stdlib.ProcessorFormatter.wrap_for_formatter, 

505 ], 

506 logger_factory=structlog.stdlib.LoggerFactory(), 

507 wrapper_class=structlog.make_filtering_bound_logger(logging.DEBUG), 

508 cache_logger_on_first_use=True, 

509 ) 

510 

511 logger = structlog.get_logger("gnssvodpy") 

512 # Ensure the underlying stdlib logger accepts all levels 

513 logging.getLogger("gnssvodpy").setLevel(logging.DEBUG) 

514 

515 _install_excepthook(logger) 

516 

517 return logger

521LOGGER = configure_logging()
