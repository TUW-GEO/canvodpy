83def _str_to_object(ds: xr.Dataset) -> xr.Dataset: 

84 """Cast StringDType / fixed-width unicode coords and vars to object dtype. 

85 

86 NumPy 2.x introduces ``StringDType`` (kind ``"T"``) which Zarr V3 / 

87 Icechunk cannot round-trip. Fixed-width unicode (kind ``"U"``) also 

88 causes dtype mismatches on store append. Normalising to ``object`` 

89 keeps behaviour identical to NumPy 1.x. 

90 """ 

91 for name in list(ds.coords) + list(ds.data_vars): 

92 if ds[name].dtype.kind in ("U", "T"): 

93 ds[name] = ds[name].astype(object) 

94 return ds

976 def parse_observation_slice( 

977 self, 

978 slice_text: str, 

979 ) -> tuple[float | None, int | None, int | None]: 

980 """Parse a RINEX observation slice into value, LLI, and SSI. 

981 

982 Enhanced to handle both standard 16-character format and 

983 variable-length records. 

984 

985 Parameters 

986 ---------- 

987 slice_text : str 

988 Observation slice to parse. 

989 

990 Returns 

991 ------- 

992 tuple[float | None, int | None, int | None] 

993 Parsed (value, LLI, SSI) tuple. 

994 

995 """ 

996 if not slice_text or not slice_text.strip(): 

997 return None, None, None 

998 

999 try: 

1000 # Method 1: Standard RINEX format with decimal at position -6 

1001 if ( 

1002 len(slice_text) >= OBS_SLICE_MIN_LEN 

1003 and len(slice_text) <= OBS_SLICE_MAX_LEN 

1004 and slice_text[OBS_SLICE_DECIMAL_POS] == "." 

1005 ): 

1006 slice_chars = list(slice_text) 

1007 ssi = slice_chars.pop(-1) if len(slice_chars) > 0 else "" 

1008 lli = slice_chars.pop(-1) if len(slice_chars) > 0 else "" 

1009 

1010 # Convert LLI and SSI 

1011 lli = int(lli) if lli.strip() and lli.isdigit() else None 

1012 ssi = int(ssi) if ssi.strip() and ssi.isdigit() else None 

1013 

1014 # Convert value 

1015 value_str = "".join(slice_chars).strip() 

1016 if value_str: 

1017 value = float(value_str) 

1018 return value, lli, ssi 

1019 

1020 except ValueError, IndexError: 

1021 pass 

1022 

1023 try: 

1024 # Method 2: Flexible parsing for variable-length records 

1025 slice_trimmed = slice_text.strip() 

1026 if not slice_trimmed: 

1027 return None, None, None 

1028 

1029 # Look for a decimal point to identify the numeric value 

1030 if "." in slice_trimmed: 

1031 # Find the main numeric value (supports negative numbers) 

1032 number_match = re.search(r"(-?\d+\.\d+)", slice_trimmed) 

1033 

1034 if number_match: 

1035 value = float(number_match.group(1)) 

1036 

1037 # Check for LLI/SSI indicators after the number 

1038 remaining_part = slice_trimmed[number_match.end() :].strip() 

1039 lli = None 

1040 ssi = None 

1041 

1042 # Parse remaining characters as potential LLI/SSI 

1043 if remaining_part: 

1044 # Could be just SSI, or LLI followed by SSI 

1045 if len(remaining_part) == 1: 

1046 # Just one indicator - assume it's SSI 

1047 if remaining_part.isdigit(): 

1048 ssi = int(remaining_part) 

1049 elif len(remaining_part) >= LLI_SSI_PAIR_LEN: 

1050 # Two or more characters - take last two as LLI, SSI 

1051 lli_char = remaining_part[-2] 

1052 ssi_char = remaining_part[-1] 

1053 

1054 if lli_char.isdigit(): 

1055 lli = int(lli_char) 

1056 if ssi_char.isdigit(): 

1057 ssi = int(ssi_char) 

1058 

1059 return value, lli, ssi 

1060 

1061 except ValueError, IndexError: 

1062 pass 

1063 

1064 # Method 3: Last resort - try simple float parsing 

1065 try: 

1066 simple_value = float(slice_text.strip()) 

1067 return simple_value, None, None 

1068 except ValueError: 

1069 pass 

1070 

1071 return None, None, None

1073 def process_satellite_data(self, s: str) -> Satellite: 

1074 """Process satellite data line into a Satellite object with observations. 

1075 

1076 Handles variable-length observation records correctly by adaptively parsing 

1077 based on the actual line length and content. 

1078 """ 

1079 sv = s[:3].strip() 

1080 satellite = Satellite(sv=sv) 

1081 bands_tbe = [f"{sv}|{b}" for b in self.header.obs_codes_per_system[sv[0]]] 

1082 

1083 # Get the data part (after sv identifier) 

1084 data_part = s[3:] 

1085 

1086 # Process each observation adaptively 

1087 for i, band in enumerate(bands_tbe): 

1088 start_idx = i * 16 

1089 end_idx = start_idx + 16 

1090 

1091 # Check if we have enough data for this observation 

1092 if start_idx >= len(data_part): 

1093 # No more data available - create empty observation 

1094 observation = Observation( 

1095 obs_type=band.split("|")[1][0], 

1096 value=None, 

1097 lli=None, 

1098 ssi=None, 

1099 ) 

1100 satellite.add_observation(observation) 

1101 continue 

1102 

1103 # Extract the slice, but handle variable length 

1104 if end_idx <= len(data_part): 

1105 # Full 16-character slice available 

1106 slice_data = data_part[start_idx:end_idx] 

1107 else: 

1108 # Partial slice - pad with spaces to maintain consistency 

1109 available_slice = data_part[start_idx:] 

1110 slice_data = available_slice.ljust(16) # Pad with spaces if needed 

1111 

1112 value, lli, ssi = self.parse_observation_slice(slice_data) 

1113 

1114 observation = Observation( 

1115 obs_type=band.split("|")[1][0], 

1116 value=value, 

1117 lli=lli, 

1118 ssi=ssi, 

1119 ) 

1120 satellite.add_observation(observation) 

1121 

1122 return satellite

1283 def infer_sampling_interval(self) -> pint.Quantity | None: 

1284 """Infer sampling interval from consecutive epoch deltas. 

1285 

1286 Returns 

1287 ------- 

1288 pint.Quantity or None 

1289 Sampling interval in seconds, or None if cannot be inferred 

1290 

1291 """ 

1292 dts = self._epoch_datetimes() 

1293 if len(dts) < MIN_EPOCHS_FOR_INTERVAL: 

1294 return None 

1295 # Compute deltas 

1296 deltas: list[timedelta] = [b - a for a, b in pairwise(dts) if b >= a] 

1297 if not deltas: 

1298 return None 

1299 # Pick the most common delta (robust to an occasional missing epoch) 

1300 seconds = Counter( 

1301 int(dt.total_seconds()) for dt in deltas if dt.total_seconds() > 0 

1302 ) 

1303 if not seconds: 

1304 return None 

1305 mode_seconds, _ = seconds.most_common(1)[0] 

1306 return (mode_seconds * UREG.second).to(UREG.seconds)

1308 def infer_dump_interval( 

1309 self, sampling_interval: pint.Quantity | None = None 

1310 ) -> pint.Quantity | None: 

1311 """Infer the intended dump interval for the RINEX file. 

1312 

1313 Parameters 

1314 ---------- 

1315 sampling_interval : pint.Quantity, optional 

1316 Known sampling interval. If provided, returns (#epochs * sampling_interval) 

1317 

1318 Returns 

1319 ------- 

1320 pint.Quantity or None 

1321 Dump interval in seconds, or None if cannot be inferred 

1322 

1323 """ 

1324 idx = self.get_epoch_record_batches() 

1325 n_epochs = len(idx) 

1326 if n_epochs == 0: 

1327 return None 

1328 

1329 if sampling_interval is not None: 

1330 return (n_epochs * sampling_interval).to(UREG.seconds) 

1331 

1332 # Fallback: time coverage inclusive (last - first) + typical step 

1333 dts = self._epoch_datetimes() 

1334 if len(dts) == 0: 

1335 return None 

1336 if len(dts) == 1: 

1337 # single epoch: treat as 1 * unknown step (cannot infer) 

1338 return None 

1339 

1340 # Estimate step from data 

1341 est_step = self.infer_sampling_interval() 

1342 if est_step is None: 

1343 return None 

1344 

1345 # Inclusive coverage often equals (n_epochs - 1) * step; intended 

1346 # dump interval is n_epochs * step. 

1347 return (n_epochs * est_step.to(UREG.seconds)).to(UREG.seconds)

1350 self,

1354 """Validate that the number of epochs matches the expected dump interval. 

1355 

1356 Parameters 

1357 ---------- 

1358 dump_interval : str or pint.Quantity, optional 

1359 Expected file dump interval. If None, inferred from epochs.

1365 None

1372 If intervals cannot be inferred

1376 if sampling_interval is None:

1378 if inferred is None:

1381 sampling_interval = inferred 

1382 # normalize to pint 

1383 elif not isinstance(sampling_interval, pint.Quantity): 

1384 sampling_interval = UREG.Quantity(sampling_interval).to(UREG.seconds) # ty: ignore[invalid-assignment]

1389 sampling_interval=sampling_interval # ty: ignore[invalid-argument-type]

1393 raise ValueError(msg)

1397 dump_interval = UREG.Quantity(dump_interval).to(UREG.seconds) # ty: ignore[invalid-assignment]

1448 def _precompute_sids_from_header( 

1449 self, 

1450 ) -> tuple[list[str], dict[str, dict[str, object]]]: 

1451 """Build sorted SID list and properties from header info alone. 

1452 

1453 Uses the header's obs_codes_per_system and static constellation 

1454 SV lists to pre-compute the full theoretical SID set, eliminating 

1455 the discovery pass. 

1456 

1457 Returns 

1458 ------- 

1459 sorted_sids : list[str] 

1460 Sorted list of signal IDs. 

1461 sid_properties : dict[str, dict[str, object]] 

1462 Mapping of SID to its properties (sv, system, band, code, 

1463 freq_center, freq_min, freq_max, bandwidth, overlapping_group). 

1464 

1465 """ 

1466 mapper = self._signal_mapper 

1467 signal_ids: set[str] = set() 

1468 sid_properties: dict[str, dict[str, object]] = {} 

1469 

1470 # Pre-compute pint arithmetic once per unique band 

1471 band_freq_cache: dict[str, tuple[float, float, float, float]] = {} 

1472 

1473 for system, obs_codes in self.header.obs_codes_per_system.items(): 

1474 svs = _get_constellation_svs(system) 

1475 

1476 for obs_code in obs_codes: 

1477 if len(obs_code) < 3: 

1478 continue 

1479 band_num = obs_code[1] 

1480 code_char = obs_code[2] 

1481 

1482 band_name = mapper.SYSTEM_BANDS.get(system, {}).get( 

1483 band_num, f"UnknownBand{band_num}" 

1484 ) 

1485 

1486 # Cache frequency arithmetic per band 

1487 if band_name not in band_freq_cache: 

1488 center_frequency = mapper.get_band_frequency(band_name) 

1489 bandwidth = mapper.get_band_bandwidth(band_name) 

1490 

1491 if center_frequency is not None and bandwidth is not None: 

1492 bw = bandwidth[0] if isinstance(bandwidth, list) else bandwidth 

1493 freq_min = center_frequency - (bw / 2.0) 

1494 freq_max = center_frequency + (bw / 2.0) 

1495 band_freq_cache[band_name] = ( 

1496 float(center_frequency), 

1497 float(freq_min), 

1498 float(freq_max), 

1499 float(bw), 

1500 ) 

1501 else: 

1502 band_freq_cache[band_name] = ( 

1503 np.nan, 

1504 np.nan, 

1505 np.nan, 

1506 np.nan, 

1507 ) 

1508 

1509 freq_center, freq_min, freq_max, bw = band_freq_cache[band_name] 

1510 overlapping_group = mapper.get_overlapping_group(band_name) 

1511 

1512 sid_suffix = "|" + band_name + "|" + code_char 

1513 

1514 for sv in svs: 

1515 sid = sv + sid_suffix 

1516 if sid not in signal_ids: 

1517 signal_ids.add(sid) 

1518 sid_properties[sid] = { 

1519 "sv": sv, 

1520 "system": system, 

1521 "band": band_name, 

1522 "code": code_char, 

1523 "freq_center": freq_center, 

1524 "freq_min": freq_min, 

1525 "freq_max": freq_max, 

1526 "bandwidth": bw, 

1527 "overlapping_group": overlapping_group, 

1528 } 

1529 

1530 sorted_sids = sorted(signal_ids) 

1531 return sorted_sids, {s: sid_properties[s] for s in sorted_sids}

1533 def _create_dataset_single_pass( 

1534 self, keep_data_vars: frozenset[str] | None = None 

1535 ) -> xr.Dataset: 

1536 """Create xarray Dataset in a single pass over the file. 

1537 

1538 Pre-allocates arrays using header-derived SID set and epoch count, 

1539 then fills them by parsing observations inline without Pydantic 

1540 models or function-call overhead. 

1541 

1542 Returns 

1543 ------- 

1544 xr.Dataset 

1545 Dataset with dimensions (epoch, sid) and standard variables. 

1546 

1547 """ 

1548 lines = self._load_file() 

1549 epoch_batches = self.get_epoch_record_batches() 

1550 n_epochs = len(epoch_batches) 

1551 

1552 sorted_sids, sid_properties = self._precompute_sids_from_header() 

1553 n_sids = len(sorted_sids) 

1554 sid_to_idx = {sid: i for i, sid in enumerate(sorted_sids)} 

1555 

1556 # Pre-allocate only the arrays that will be kept 

1557 _all = keep_data_vars is None 

1558 need_snr = _all or "SNR" in keep_data_vars 

1559 need_pseudo = _all or "Pseudorange" in keep_data_vars 

1560 need_phase = _all or "Phase" in keep_data_vars 

1561 need_doppler = _all or "Doppler" in keep_data_vars 

1562 need_lli = _all or "LLI" in keep_data_vars 

1563 need_ssi = _all or "SSI" in keep_data_vars 

1564 

1565 timestamps = np.empty(n_epochs, dtype="datetime64[ns]") 

1566 if need_snr: 

1567 snr = np.full((n_epochs, n_sids), np.nan, dtype=DTYPES["SNR"]) 

1568 if need_pseudo: 

1569 pseudo = np.full((n_epochs, n_sids), np.nan, dtype=DTYPES["Pseudorange"]) 

1570 if need_phase: 

1571 phase = np.full((n_epochs, n_sids), np.nan, dtype=DTYPES["Phase"]) 

1572 if need_doppler: 

1573 doppler = np.full((n_epochs, n_sids), np.nan, dtype=DTYPES["Doppler"]) 

1574 if need_lli: 

1575 lli = np.full((n_epochs, n_sids), -1, dtype=DTYPES["LLI"]) 

1576 if need_ssi: 

1577 ssi = np.full((n_epochs, n_sids), -1, dtype=DTYPES["SSI"]) 

1578 

1579 # Build obs_code → (obs_type, sid_suffix) lookup per system 

1580 mapper = self._signal_mapper 

1581 system_obs_lut: dict[str, list[tuple[str, str]]] = {} 

1582 for system, obs_codes in self.header.obs_codes_per_system.items(): 

1583 lut: list[tuple[str, str]] = [] 

1584 for obs_code in obs_codes: 

1585 if len(obs_code) < 3: 

1586 lut.append(("", "")) 

1587 continue 

1588 band_num = obs_code[1] 

1589 code_char = obs_code[2] 

1590 band_name = mapper.SYSTEM_BANDS.get(system, {}).get( 

1591 band_num, f"UnknownBand{band_num}" 

1592 ) 

1593 obs_type = obs_code[0] 

1594 lut.append((obs_type, "|" + band_name + "|" + code_char)) 

1595 system_obs_lut[system] = lut 

1596 

1597 # Single pass over all epochs — skip unparseable epoch lines 

1598 valid_mask = np.ones(n_epochs, dtype=bool) 

1599 for t_idx, (start, end) in enumerate(epoch_batches): 

1600 epoch_line = lines[start] 

1601 

1602 # Inline epoch parsing (no Pydantic model) 

1603 m = _EPOCH_RE.match(epoch_line) 

1604 if m is None: 

1605 valid_mask[t_idx] = False 

1606 continue 

1607 

1608 year, month, day = int(m[1]), int(m[2]), int(m[3]) 

1609 hour, minute = int(m[4]), int(m[5]) 

1610 seconds = float(m[6]) 

1611 sec_int = int(seconds) 

1612 usec = int((seconds - sec_int) * 1_000_000) 

1613 ts = np.datetime64( 

1614 f"{year:04d}-{month:02d}-{day:02d}" 

1615 f"T{hour:02d}:{minute:02d}:{sec_int:02d}", 

1616 "ns", 

1617 ) 

1618 ts += np.timedelta64(usec, "us") 

1619 timestamps[t_idx] = ts 

1620 

1621 # Parse satellite data lines inline 

1622 for line_idx in range(start + 1, end): 

1623 sat_line = lines[line_idx] 

1624 if len(sat_line) < 3: 

1625 continue 

1626 sv = sat_line[:3].strip() 

1627 if not sv: 

1628 continue 

1629 system = sv[0] 

1630 lut_list = system_obs_lut.get(system) 

1631 if lut_list is None: 

1632 continue 

1633 

1634 data_part = sat_line[3:] 

1635 data_part_len = len(data_part) 

1636 

1637 for i, (obs_type, sid_suffix) in enumerate(lut_list): 

1638 if not obs_type: 

1639 continue 

1640 

1641 col_start = i * 16 

1642 if col_start >= data_part_len: 

1643 break 

1644 

1645 sid_key = sv + sid_suffix 

1646 s_idx = sid_to_idx.get(sid_key) 

1647 if s_idx is None: 

1648 continue 

1649 

1650 col_end = col_start + 16 

1651 slice_text = data_part[col_start:col_end] 

1652 

1653 value, obs_lli, obs_ssi = _parse_obs_fast(slice_text) 

1654 if value is None: 

1655 continue 

1656 

1657 match obs_type: 

1658 case "S": 

1659 if need_snr and value != 0: 

1660 snr[t_idx, s_idx] = value 

1661 case "C": 

1662 if need_pseudo: 

1663 pseudo[t_idx, s_idx] = value 

1664 case "L": 

1665 if need_phase: 

1666 phase[t_idx, s_idx] = value 

1667 case "D": 

1668 if need_doppler: 

1669 doppler[t_idx, s_idx] = value 

1670 

1671 if need_lli and obs_lli is not None: 

1672 lli[t_idx, s_idx] = obs_lli 

1673 if need_ssi and obs_ssi is not None: 

1674 ssi[t_idx, s_idx] = obs_ssi 

1675 

1676 # Drop epochs that failed to parse 

1677 if not valid_mask.all(): 

1678 timestamps = timestamps[valid_mask] 

1679 if need_snr: 

1680 snr = snr[valid_mask] 

1681 if need_pseudo: 

1682 pseudo = pseudo[valid_mask] 

1683 if need_phase: 

1684 phase = phase[valid_mask] 

1685 if need_doppler: 

1686 doppler = doppler[valid_mask] 

1687 if need_lli: 

1688 lli = lli[valid_mask] 

1689 if need_ssi: 

1690 ssi = ssi[valid_mask] 

1691 

1692 # Build coordinate arrays from pre-computed properties 

1693 sv_list = np.array( 

1694 [sid_properties[sid]["sv"] for sid in sorted_sids], dtype=object 

1695 ) 

1696 constellation_list = np.array( 

1697 [sid_properties[sid]["system"] for sid in sorted_sids], dtype=object 

1698 ) 

1699 band_list = np.array( 

1700 [sid_properties[sid]["band"] for sid in sorted_sids], dtype=object 

1701 ) 

1702 code_list = np.array( 

1703 [sid_properties[sid]["code"] for sid in sorted_sids], dtype=object 

1704 ) 

1705 freq_center_list = [sid_properties[sid]["freq_center"] for sid in sorted_sids] 

1706 freq_min_list = [sid_properties[sid]["freq_min"] for sid in sorted_sids] 

1707 freq_max_list = [sid_properties[sid]["freq_max"] for sid in sorted_sids] 

1708 

1709 signal_id_coord = xr.DataArray( 

1710 np.array(sorted_sids, dtype=object), 

1711 dims=["sid"], 

1712 attrs=COORDS_METADATA["sid"], 

1713 ) 

1714 coords = { 

1715 "epoch": ("epoch", timestamps, COORDS_METADATA["epoch"]), 

1716 "sid": signal_id_coord, 

1717 "sv": ("sid", sv_list, COORDS_METADATA["sv"]), 

1718 "system": ("sid", constellation_list, COORDS_METADATA["system"]), 

1719 "band": ("sid", band_list, COORDS_METADATA["band"]), 

1720 "code": ("sid", code_list, COORDS_METADATA["code"]), 

1721 "freq_center": ( 

1722 "sid", 

1723 np.asarray(freq_center_list, dtype=DTYPES["freq_center"]), 

1724 COORDS_METADATA["freq_center"], 

1725 ), 

1726 "freq_min": ( 

1727 "sid", 

1728 np.asarray(freq_min_list, dtype=DTYPES["freq_min"]), 

1729 COORDS_METADATA["freq_min"], 

1730 ), 

1731 "freq_max": ( 

1732 "sid", 

1733 np.asarray(freq_max_list, dtype=DTYPES["freq_max"]), 

1734 COORDS_METADATA["freq_max"], 

1735 ), 

1736 } 

1737 

1738 if self.header.signal_strength_unit == UREG.dBHz: 

1739 snr_meta = CN0_METADATA 

1740 else: 

1741 snr_meta = SNR_METADATA 

1742 

1743 _data_vars: dict = {} 

1744 if need_snr: 

1745 _data_vars["SNR"] = (["epoch", "sid"], snr, snr_meta) 

1746 if need_pseudo: 

1747 _data_vars["Pseudorange"] = ( 

1748 ["epoch", "sid"], 

1749 pseudo, 

1750 OBSERVABLES_METADATA["Pseudorange"], 

1751 ) 

1752 if need_phase: 

1753 _data_vars["Phase"] = ( 

1754 ["epoch", "sid"], 

1755 phase, 

1756 OBSERVABLES_METADATA["Phase"], 

1757 ) 

1758 if need_doppler: 

1759 _data_vars["Doppler"] = ( 

1760 ["epoch", "sid"], 

1761 doppler, 

1762 OBSERVABLES_METADATA["Doppler"], 

1763 ) 

1764 if need_lli: 

1765 _data_vars["LLI"] = (["epoch", "sid"], lli, OBSERVABLES_METADATA["LLI"]) 

1766 if need_ssi: 

1767 _data_vars["SSI"] = (["epoch", "sid"], ssi, OBSERVABLES_METADATA["SSI"]) 

1768 

1769 ds = xr.Dataset( 

1770 data_vars=_data_vars, 

1771 coords=coords, 

1772 attrs={**self._build_attrs()}, 

1773 ) 

1774 

1775 if self.apply_overlap_filter: 

1776 ds = self.filter_by_overlapping_groups(ds, self.overlap_preferences) 

1777 

1778 return ds
