152 None, 

153 description=( 

154 "Network-wide, fingerprint-keyed aux (SP3/CLK Hermite) cache " 

155 "root (dev/todo_later.md §44). None (default) = disabled, falls " 

156 "back to per-site aux_data_dir/aux_{date}.zarr, rebuilt every " 

157 "run. Ephemeris products are satellite-based, not site-based, " 

158 "so pointing multiple sites at the same path lets them share " 

159 "one cache entry per (agency, product, date) instead of each " 

160 "rebuilding it independently." 

161 ), 

162 ) 

163 gnss_store_strategy: Literal["skip", "overwrite", "unsafe_append"] = Field( 

164 "skip", 

165 description=( 

166 "How to handle a file whose hash/time-range already exists in " 

167 "the GNSS store (new files are always written regardless of " 

168 "this setting). 'skip' (default, recommended): do nothing -- " 

169 "the normal case, since a re-run over already-ingested files " 

170 "should be a no-op, not a rewrite. 'overwrite': delete the old " 

171 "epochs for that file's range first, then write the new " 

172 "version -- for correcting already-ingested data after a " 

173 "pipeline bug fix (reader logic, ephemeris source, etc.), not " 

174 "because the raw GNSS file itself changed. 'unsafe_append': " 

175 "write the file's data again on top of what's already there, " 

176 "with NO epoch-level uniqueness check. This produces duplicate " 

177 "epoch coordinate values in the Zarr array, and xarray does not " 

178 "enforce unique index values. The two built-in pipeline read " 

179 "paths already guard against it -- GnssResearchSite." 

180 "read_receiver_data() routes through read_group_deduplicated() " 

181 "whenever this strategy is set (manager.py), and VodComputer." 

182 "compute_bulk() unconditionally deduplicates every read via " 

183 "_dedup_sort() regardless of strategy -- but MyIcechunkStore." 

184 "read_group() itself (the low-level API used directly by " 

185 "custom scripts/notebooks, and by the deprecated L1 " 

186 "Pipeline.calculate_vod()) has no such guard: `.sel()` on a " 

187 "duplicated label can raise or return multiple matches, and " 

188 "aligning two duplicated datasets (e.g. canopy vs. reference) " 

189 "produces a cartesian product at the duplicated epochs. The " 

190 "duplication is also physical, not just a read-time artifact -- " 

191 "Icechunk stores the extra chunks permanently; dedup-on-read " 

192 "does not undo that. Avoid 'unsafe_append' unless you have a " 

193 "specific reason, and be aware any read path outside the two " 

194 "guarded ones above is unprotected." 

195 ), 

196 ) 

197 vod_store_strategy: Literal["skip", "overwrite", "unsafe_append"] = "overwrite"

210 @field_validator("stores_root_dir", mode="before") 

211 @classmethod 

212 def validate_stores_dir(cls, v: object) -> object: 

213 """Expand ~ and reject placeholder values in stores_root_dir.""" 

214 if isinstance(v, str): 

215 if v.strip() in {"/path/to/stores", "/path/to/your/stores"}: 

216 raise ValueError( 

217 f"stores_root_dir is set to placeholder {v!r} — " 

218 "set a real directory path in canvod-settings.yaml" 

219 ) 

220 return Path(v).expanduser() 

221 return v
