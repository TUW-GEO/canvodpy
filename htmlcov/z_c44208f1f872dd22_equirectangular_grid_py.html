10class EquirectangularBuilder(BaseGridBuilder): 

11 """Simple rectangular grid in (theta, phi) space. 

12 

13 The hemisphere is divided into a regular rectangular array: a constant 

14 number of theta bands, each containing the same constant number of phi 

15 sectors. Every cell is a rectangle of the same angular size in 

16 (theta, phi), except the last band and the last sector, which are 

17 narrower when Δθ does not divide their range. It differs from 

18 ``EqualAngleBuilder`` at the zenith: ``EqualAngleBuilder`` makes the 

19 innermost half band a single zenith cap and puts its band edges half a 

20 band off, while this builder starts the first band at the zenith with 

21 the same number of sectors as every other band. 

22 

23 A cell's solid angle is Δφ · (cos θ_min − cos θ_max) ≈ Δφ · Δθ · sin θ, 

24 so cells near the zenith subtend *less* solid angle than cells near the 

25 horizon. A statistic that weights every cell equally therefore gives 

26 the sky near the zenith more weight than its share of the hemisphere. 

27 **Not recommended for scientific analysis** – use ``EqualAreaBuilder`` 

28 instead. 

29 

30 Coordinate convention (physics / GNSS) 

31 --------------------------------------- 

32 * phi ∈ [0, 2π) – azimuthal angle from North, clockwise (navigation convention) 

33 * theta ∈ [0, π/2] – polar angle from zenith (0 = straight up, 

34 π/2 = horizon) 

35 

36 What ``angular_resolution`` means 

37 ---------------------------------- 

38 ``angular_resolution`` (degrees) is used as **both** the theta-band width 

39 *and* the phi-sector width:: 

40 

41 n_theta = ceil((π/2 − cutoff) / Δθ) 

42 n_phi = ceil(2π / Δθ) 

43 total cells = n_theta × n_phi 

44 

45 Mathematical construction 

46 ------------------------- 

47 1. Theta edges are placed at ``0``, ``Δθ``, ``2Δθ``, … up to 

48 ``π/2 − cutoff_theta`` (the last edge is clamped exactly to 

49 ``π/2 − cutoff_theta`` rather than overshooting it). 

50 2. Phi edges are placed at 0, Δθ, 2Δθ, … and a last edge at 2π; the 

51 last sector is narrower when Δθ does not divide 360°. 

52 3. Every (theta_band, phi_sector) combination produces one cell. The 

53 cell center is the midpoint of the rectangle. 

54 4. No special zenith cap is created; the band nearest the zenith has 

55 the same number of phi sectors as all other bands. 

56 

57 Parameters 

58 ---------- 

59 angular_resolution : float 

60 Angular spacing in degrees, applied identically in both theta and phi. 

61 cutoff_theta : float 

62 Elevation mask angle in degrees. The last band ends at 

63 θ = 90° − cutoff_theta; the sky below that elevation gets no cells. 

64 phi_rotation : float 

65 Rigid azimuthal rotation applied after construction, in degrees. 

66 

67 """

80 def _build_grid( 

81 self, 

82 ) -> tuple[pl.DataFrame, np.ndarray, list[np.ndarray], list[np.ndarray]]: 

83 """Construct the equirectangular hemisphere grid. 

84 

85 Returns 

86 ------- 

87 grid : pl.DataFrame 

88 One row per cell with columns: phi, theta, phi_min, phi_max, 

89 theta_min, theta_max, cell_id. 

90 theta_lims : np.ndarray 

91 Outer theta edge of each band (radians). 

92 phi_lims : list[np.ndarray] 

93 Array of phi_min values for each band (identical across bands). 

94 cell_ids : list[np.ndarray] 

95 Cell-id arrays, one per band. 

96 

97 """ 

98 # Elevation mask: keep bands from the zenith (theta=0) out to 

99 # (pi/2 - cutoff_theta), i.e. drop the near-horizon band(s) -- 

100 # matching EqualAreaBuilder/EqualAngleBuilder convention. Anchor the 

101 # last edge exactly at max_theta (rather than letting arange's step 

102 # overshoot it) so no band extends past the horizon or past the 

103 # cutoff when angular_resolution doesn't evenly divide it. 

104 max_theta = np.pi / 2 - self.cutoff_theta_rad 

105 

106 theta_edges = np.arange(0, max_theta, self.angular_resolution_rad) 

107 theta_edges = np.append(theta_edges, max_theta) 

108 if len(theta_edges) > 1 and np.isclose(theta_edges[-1], theta_edges[-2]): 

109 theta_edges = theta_edges[:-1] 

110 

111 # The same for the sectors: the last edge is exactly 2*pi, so the last 

112 # sector (narrower when angular_resolution doesn't divide 360 deg) 

113 # has its center inside it, and rounding in arange adds no 

114 # zero-width sector at 2*pi. 

115 phi_edges = np.arange(0, 2 * np.pi, self.angular_resolution_rad) 

116 phi_edges = np.append(phi_edges, 2 * np.pi) 

117 if np.isclose(phi_edges[-1], phi_edges[-2]): 

118 phi_edges = np.delete(phi_edges, -2) 

119 

120 cells = [] 

121 

122 for i in range(len(theta_edges) - 1): 

123 theta_min, theta_max = theta_edges[i], theta_edges[i + 1] 

124 

125 for j in range(len(phi_edges) - 1): 

126 phi_min, phi_max = phi_edges[j], phi_edges[j + 1] 

127 

128 cells.append( 

129 { 

130 "phi": (phi_min + phi_max) / 2, 

131 "theta": (theta_min + theta_max) / 2, 

132 "phi_min": phi_min, 

133 "phi_max": phi_max, 

134 "theta_min": theta_min, 

135 "theta_max": theta_max, 

136 } 

137 ) 

138 

139 grid = pl.DataFrame(cells).with_columns( 

140 pl.int_range(0, pl.len()).alias("cell_id") 

141 ) 

142 

143 theta_lims = theta_edges[1:] 

144 phi_lims = [phi_edges[:-1] for _ in range(len(theta_edges) - 1)] 

145 cell_ids_list = [ 

146 np.arange(i * (len(phi_edges) - 1), (i + 1) * (len(phi_edges) - 1)) 

147 for i in range(len(theta_edges) - 1) 

148 ] 

149 

150 return grid, theta_lims, phi_lims, cell_ids_list
