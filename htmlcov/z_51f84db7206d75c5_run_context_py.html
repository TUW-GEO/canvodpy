1"""Per-run identifier propagation for logging. 

2 

3A run_id identifies a single pipeline invocation for one site (see 

4``cli/run.py``'s per-site loop). It is threaded automatically into every 

5log record via a structlog processor (``logging_config.py``) and, where 

6possible, into Icechunk commit messages, so a human or an agent can 

7correlate "this run" across the human log, the agent-diagnostic log, and 

8the data it wrote -- without any call site needing to pass it explicitly. 

9""" 

10 

11import contextvars 

12 

13RUN_ID: contextvars.ContextVar[str | None] = contextvars.ContextVar( 

14 "RUN_ID", default=None 

15) 

16 

17 

18def get_run_id() -> str | None: 

19 """Return the run_id bound to the current context, if any.""" 

20 return RUN_ID.get() 

21 

22 

23def set_run_id(run_id: str) -> contextvars.Token: 

24 """Bind ``run_id`` to the current context. 

25 

26 Parameters 

27 ---------- 

28 run_id : str 

29 Identifier for the current pipeline run (see ``cli/run.py``'s 

30 ``{site}-{YYYYMMDD-HHMMSS}`` format). 

31 

32 Returns 

33 ------- 

34 contextvars.Token 

35 Token used to restore the previous value via ``reset_run_id``. 

36 """ 

37 return RUN_ID.set(run_id)

39 

40def reset_run_id(token: contextvars.Token) -> None: 

41 """Reset RUN_ID to its previous value."""
