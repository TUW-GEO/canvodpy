1"""Entries of the store history (``summaries.history`` of the store metadata). 

2 

3Deprecated: the history of a store is its commit log (ADR 0014), and 

4canVODpy no longer writes ``summaries.history``. 

5""" 

6 

7from __future__ import annotations 

8 

9import importlib.metadata 

10 

11from canvod.utils.tools import deprecated 

12 

13 

14@deprecated( 

15 "history_entry() is left over from development and will be removed with the " 

16 "next major version. The history of a store is its commit log; " 

17 "canvod.store_metadata.show_metadata lists it." 

18) 

19def history_entry(now: str, event: str, preprocessing: dict[str, str]) -> str: 

20 """One history line: time, event, canVODpy version and preprocessing. 

21 

22 Parameters 

23 ---------- 

24 now : str 

25 ISO 8601 time of the write. 

26 event : str 

27 What was written, e.g. ``"Ingest 2025001: canopy_01(written=96)"``. 

28 preprocessing : dict[str, str] 

29 Per store group written, its preprocessing as text 

30 (``canvod.config.models.describe_preprocessing``). 

31 """ 

32 version = importlib.metadata.version("canvodpy") 

33 applied = "; ".join(f"{group}: {text}" for group, text in preprocessing.items()) 

34 return f"{now}: {event} (canvodpy {version}); preprocessing: {applied or 'none'}"
