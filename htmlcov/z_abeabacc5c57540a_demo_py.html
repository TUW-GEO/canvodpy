1"""Demo data: one day of the Rosalia site, downloaded on first use. 

2 

3``demo_data()`` downloads the published archives of one kind of data, 

4checks each archive's sha256, unpacks it into a cache folder and returns 

5that folder. The folder has the layout of the test data, so the demo 

6notebooks and the tests find the files at the same relative paths. A 

7second call finds the data in the cache and downloads nothing. 

8 

9The archives (ADR 0017): 

10 

11- RINEX v3.04, the whole of 2025-01-01, canopy and reference receiver; 

12- SBF, 00:00 to 02:00 of 2025-01-01, canopy and reference receiver; 

13- auxiliary files: SP3 orbits and CLK clocks of 2024-12-31 to 2025-01-02 

14 and the auxiliary store of 2025-01-01. 

15 

16The kind ``"stores"`` downloads the GNSS store and the VOD store of the 

17RINEX v3.04 day from the published reference stores (spec #318): the 

18Hugging Face dataset and commit named in ``reference-stores/published.yaml`` 

19of this package, checked against ``reference-stores/SHA256SUMS`` next to 

20it. ``just reference-check`` reads the same two files 

21(:func:`published_reference`) but unpacks into a folder of its own, so a 

22demo store that is written to never changes what the check compares with. 

23 

24``ARCHIVES`` is the single list of what each archive holds; 

25``just demo-archives`` builds the zips from it. The sha256 values are 

26those of zips built from the test data at commit 2024c42 of 

27``packages/canvod-readers/tests/test_data`` (the run history at commit 

28f0764c45); its releases are archived on Zenodo 

29(10.5281/zenodo.19708759). The zips are in the Hugging Face dataset 

30``nfb2021/canvodpy-demo-data``; the URLs name one commit of it, so the 

31files can't change under a released canvodpy. 

32"""

175def read_sums(path: Path) -> dict[str, str]: 

176 """The sha256 of every file in a ``SHA256SUMS`` file, by its path.""" 

177 sums = {} 

178 for line in path.read_text(encoding="utf-8").splitlines(): 

179 if line.strip(): 

180 sha256, name = line.split(maxsplit=1) 

181 sums[name.strip()] = sha256 

182 return sums

217def demo_data(kind: DemoKind, *, cache_dir: Path | None = None) -> Path: 

218 """Download and unpack one kind of demo data. 

219 

220 Parameters 

221 ---------- 

222 kind : {"rinex", "sbf", "aux", "run_history", "stores"} 

223 ``"rinex"``: RINEX v3.04 of both receivers, the whole day. 

224 ``"sbf"``: SBF of both receivers, 00:00 to 02:00. 

225 ``"aux"``: SP3 and CLK files and the auxiliary store. 

226 ``"run_history"``: an anonymized run history of nine runs, a log 

227 folder for the run dashboard (``<folder>/run_history``). 

228 ``"stores"``: the GNSS store and the VOD store of the Rosalia 

229 RINEX v3.04 day, 2025-01-01, as canVODpy writes them: the 

230 published reference stores of the profile ``default``. 

231 cache_dir : Path, optional 

232 Where to keep the data. Default: the user cache folder of 

233 canvodpy (``default_cache_dir()``). 

234 

235 Returns 

236 ------- 

237 Path 

238 The folder that holds the files. For the raw data, the auxiliary 

239 files and the run history it is one folder, in the layout of the 

240 test data, for example 

241 ``<folder>/rinex_v3_04/01_Rosalia/02_canopy/...``. For 

242 ``"stores"`` it is a folder of the published revision of the 

243 reference stores, with the stores in 

244 ``<folder>/stores/rosalia_rinex_v3_04/gnss`` and ``.../vod``. 

245 They are unpacked once and not checked again: copy them before 

246 writing to them, so the next call returns them unchanged. 

247 

248 Raises 

249 ------ 

250 ValueError 

251 If ``kind`` is not one of the known ones. 

252 RuntimeError 

253 If an archive is not published yet, can't be downloaded, or its 

254 sha256 doesn't match. 

255 

256 Examples 

257 -------- 

258 >>> from canvodpy import demo_data 

259 >>> root = demo_data("rinex") # doctest: +SKIP 

260 >>> gnss = demo_data("stores") / "stores/rosalia_rinex_v3_04/gnss" # doctest: +SKIP 

261 """ 

262 if kind == "stores": 

263 return _reference_stores(cache_dir) 

264 archives = [a for a in ARCHIVES if a.kind == kind] 

265 if not archives: 

266 kinds = {a.kind for a in ARCHIVES} | {"stores"} 

267 known = ", ".join(repr(t) for t in sorted(kinds)) 

268 msg = f"Unknown kind of demo data {kind!r}; known: {known}" 

269 raise ValueError(msg) 

270 

271 root = Path(cache_dir or default_cache_dir()) / DEMO_DATA_VERSION 

272 for archive in archives: 

273 _ensure(archive, root) 

274 return root

277def _reference_stores(cache_dir: Path | None) -> Path: 

278 """Download and unpack the GNSS and VOD store of the RINEX v3.04 day.""" 

279 published = published_reference() 

280 root = Path(cache_dir or default_cache_dir()) / "stores" / published.revision 

281 for file in _STORE_ZIPS: 

282 name = f"default/{file}" 

283 fetch_archive(file, published.url(name), published.sums[name], root) 

284 return root

287def _download(url: str, path: Path) -> None: 

288 """Download *url* to *path*, continuing after a broken connection. 

289 

290 The bytes received so far are kept in ``<path>.part``; each attempt 

291 asks for the rest (an HTTP Range request), also 

292 in a later call after this one gave up. 

293 

294 Raises 

295 ------ 

296 OSError 

297 If the download is still incomplete after the last attempt (the 

298 error of that attempt). 

299 """ 

300 part = path.with_name(path.name + ".part") 

301 part.parent.mkdir(parents=True, exist_ok=True) 

302 error: OSError | None = None 

303 for attempt in range(DOWNLOAD_ATTEMPTS): 

304 if attempt: 

305 time.sleep(min(2**attempt, 30)) 

306 have = part.stat().st_size if part.exists() else 0 

307 request = urllib.request.Request(url) 

308 if have: 

309 request.add_header("Range", f"bytes={have}-") 

310 try: 

311 with urllib.request.urlopen(request, timeout=60) as response: 

312 if response.status != 206: 

313 have = 0 # the server sends the whole file again 

314 total = _total_size(response, have) 

315 with part.open("ab" if have else "wb") as f: 

316 shutil.copyfileobj(response, f, 1 << 20) 

317 except urllib.error.HTTPError as e: 

318 if e.code < 500: 

319 raise # not found, or not allowed: trying again won't help 

320 error = e 

321 continue 

322 except (OSError, http.client.HTTPException) as e: 

323 error = e if isinstance(e, OSError) else OSError(str(e)) 

324 continue 

325 if total is None or part.stat().st_size == total: 

326 part.replace(path) 

327 return 

328 error = OSError(f"got {part.stat().st_size} of {total} bytes") 

329 assert error is not None 

330 raise error

333def _total_size(response: http.client.HTTPResponse, start: int) -> int | None: 

334 """The size of the whole file, from the headers of *response*.""" 

335 if content_range := response.headers.get("Content-Range"): 

336 return int(content_range.rsplit("/", 1)[1]) 

337 if length := response.headers.get("Content-Length"): 

338 return start + int(length) 

339 return None

342def _ensure(archive: DemoArchive, root: Path) -> None: 

343 """Download, check and unpack one archive, unless it is in the cache.""" 

344 if archive.url is None: 

345 marker = root / f".{archive.name}.sha256" 

346 if marker.exists() and marker.read_text().strip() == archive.sha256: 

347 return 

348 msg = f"Demo archive {archive.name} is not published yet" 

349 raise RuntimeError(msg) 

350 fetch_archive(archive.name, archive.url, archive.sha256, root)

353def fetch_archive(name: str, url: str, sha256: str, root: Path) -> None: 

354 """Download a zip, check its sha256 and unpack it into ``root``. 

355 

356 A marker file in ``root`` records the sha256 of the unpacked zip; a 

357 second call with the same sha256 downloads nothing. The download 

358 continues after a broken connection (``_download``). Used for the 

359 demo data and the reference stores (``scripts/reference_stores.py``). 

360 

361 Parameters 

362 ---------- 

363 name : str 

364 File name of the zip; names the marker file. 

365 url : str 

366 Where to download it; a ``file://`` URL reads a local zip. 

367 sha256 : str 

368 The expected sha256 of the zip. 

369 root : Path 

370 The folder to unpack into. 

371 

372 Raises 

373 ------ 

374 RuntimeError 

375 If the zip can't be downloaded or its sha256 doesn't match; it is 

376 then not unpacked. 

377 """ 

378 marker = root / f".{name}.sha256" 

379 if marker.exists() and marker.read_text().strip() == sha256: 

380 return 

381 

382 zip_path = root / ".downloads" / name 

383 try: 

384 _download(url, zip_path) 

385 except OSError as e: 

386 msg = ( 

387 f"Could not download archive {name} from {url}: {e}. Check the " 

388 "connection and try again; a new attempt continues where this " 

389 "one stopped." 

390 ) 

391 raise RuntimeError(msg) from e 

392 

393 try: 

394 with zip_path.open("rb") as f: 

395 found = hashlib.file_digest(f, "sha256").hexdigest() 

396 if found != sha256: 

397 msg = ( 

398 f"Archive {name} has sha256 {found}, expected {sha256}; it " 

399 "was not unpacked" 

400 ) 

401 raise RuntimeError(msg) 

402 with zipfile.ZipFile(zip_path) as zf: 

403 zf.extractall(root) 

404 finally: 

405 zip_path.unlink(missing_ok=True) 

406 marker.write_text(sha256)
