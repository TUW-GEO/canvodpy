1"""Fluent workflow API with deferred execution. 

2 

3.. deprecated:: 

4 Use ``Site(site).pipeline()`` to process data and ``Site(site).vod`` 

5 to compute VOD. Removed with the next major version. 

6 ``FluentWorkflow`` emits a ``FutureWarning`` on instantiation. 

7 

8Provides a chainable, lazy pipeline where steps are recorded and 

9executed only when a terminal method is called. 

10 

11Examples 

12-------- 

13Process RINEX data and compute VOD: 

14 

15 >>> import canvodpy 

16 >>> result = (canvodpy.workflow("ExampleSite") 

17 ... .read("2025001") 

18 ... .preprocess(agency="COD") 

19 ... .grid("equal_area", angular_resolution=5.0) 

20 ... .vod("canopy_01", "reference_01") 

21 ... .result()) 

22 

23Preview the execution plan without running it: 

24 

25 >>> plan = (canvodpy.workflow("ExampleSite") 

26 ... .read("2025001") 

27 ... .preprocess() 

28 ... .grid() 

29 ... .vod("canopy_01", "reference_01") 

30 ... .explain()) 

31 

32"""

64def _concat_epoch_datasets(datasets: list) -> Any: 

65 """Concatenate xarray Datasets along 'epoch' without xr.concat overhead. 

66 

67 Direct numpy concatenation bypasses xarray's alignment checks and 

68 intermediate copies. Safe when all datasets share the same 'sid' axis, 

69 which is guaranteed after pad_to_global_sid=True (the default). 

70 """ 

71 import numpy as np 

72 import xarray as xr 

73 

74 if len(datasets) == 1: 

75 return datasets[0] 

76 

77 first = datasets[0] 

78 merged_vars: dict = {} 

79 for var in first.data_vars: 

80 epoch_ax = list(first[var].dims).index("epoch") 

81 merged_vars[var] = ( 

82 first[var].dims, 

83 np.concatenate([ds[var].values for ds in datasets], axis=epoch_ax), 

84 ) 

85 

86 coords: dict = {} 

87 for k, coord in first.coords.items(): 

88 if "epoch" in coord.dims: 

89 ax = list(coord.dims).index("epoch") 

90 coords[k] = ( 

91 coord.dims, 

92 np.concatenate([ds.coords[k].values for ds in datasets], axis=ax), 

93 ) 

94 else: 

95 coords[k] = coord 

96 

97 return xr.Dataset(merged_vars, coords=coords, attrs=first.attrs)

105def step(method): 

106 """Record a pipeline step for deferred execution. 

107 

108 The decorated method is not called immediately. Instead, a reference 

109 to the method and its arguments is appended to ``self._plan``. 

110 The method returns ``self`` so calls can be chained. 

111 """ 

112 

113 @wraps(method) 

114 def wrapper(self, *args, **kwargs): 

115 self._plan.append((method, args, kwargs)) 

116 return self 

117 

118 return wrapper 

119 

120 

121def terminal(method): 

122 """Execute all recorded steps, then run the terminal method. 

123 

124 Iterates over ``self._plan``, calling each recorded step in order, 

125 then invokes the decorated method and returns its result. The plan 

126 is cleared after execution so the workflow can be reused. 

127 """ 

128 

129 @wraps(method) 

130 def wrapper(self, *args, **kwargs): 

131 for fn, a, kw in self._plan: 

132 fn(self, *a, **kw) 

133 self._plan.clear() 

134 return method(self, *args, **kwargs) 

135 

136 return wrapper

144@deprecated( 

145 "FluentWorkflow is left over from development and will be removed with the next major version. "

149class FluentWorkflow: 

150 """Chainable, deferred-execution workflow for VOD analysis. 

151 

152 Parameters 

153 ---------- 

154 site : str or Site 

155 Site name or :class:`~canvodpy.api.Site` object. 

156 reader : str 

157 Registered reader name (default ``"rinex3"``). 

158 grid_type : str 

159 Registered grid type (default ``"equal_area"``). 

160 vod_calculator : str 

161 Registered VOD calculator (default ``"tau_omega"``). 

162 keep_vars : list[str], optional 

163 RINEX variables to retain. Defaults to :data:`KEEP_RNX_VARS`. 

164 """ 

165 

166 def __init__( 

167 self, 

168 site: str | Site, 

169 reader: str = "rinex3", 

170 grid_type: str = "equal_area", 

171 vod_calculator: str = "tau_omega", 

172 keep_vars: list[str] | None = None, 

173 ) -> None: 

174 self._plan: list[tuple] = [] 

175 

176 # State populated by steps 

177 self._datasets: dict[str, xr.Dataset] = {} 

178 self._vod_result: xr.Dataset | None = None 

179 self._vod_pair: tuple[str, str] | None = None 

180 self._grid: Any = None 

181 self._last_date: str | None = None 

182 

183 # Configuration 

184 self._site = Site(site) if isinstance(site, str) else site 

185 self._reader_name = reader 

186 self._grid_type = grid_type 

187 self._vod_calculator_name = vod_calculator 

188 if keep_vars is None: 

189 from canvod.config import load_config 

190 

191 keep_vars = load_config().processing.params.keep_gnss_observables 

192 self._keep_vars = keep_vars 

193 

194 self.log = structlog.get_logger(__name__).bind(site=self._site.name)

200 @step 

201 def read(self, date: str, receivers: list[str] | None = None) -> FluentWorkflow: 

202 """Load the observations of *date*. 

203 

204 Files are found as ``canvodpy run`` finds them (see 

205 :mod:`canvodpy.orchestrator.discovery`): the receiver's naming recipe 

206 if it has one, otherwise canonical canVOD names only, anywhere below 

207 the receiver's directory. 

208 

209 Parameters 

210 ---------- 

211 date : str 

212 Date in ``YYYYDOY`` format (e.g. ``"2025001"``). 

213 receivers : list[str], optional 

214 Receiver names to load. If ``None``, all active receivers 

215 for the site are loaded. 

216 

217 Raises 

218 ------ 

219 ValueError 

220 If files cannot be selected for the workflow's reader. 

221 """ 

222 self._last_date = date 

223 receiver_list = receivers or list(self._site.active_receivers.keys()) 

224 log = self.log.bind(date=date) 

225 

226 from canvod.config import load_config 

227 from canvodpy.orchestrator.processor import SOURCE_FORMAT_ATTR 

228 from canvodpy.workflows.tasks import _day_files, _resolve_date 

229 

230 reader_format = _READER_FILE_FORMATS.get(self._reader_name) 

231 if reader_format is None: 

232 msg = ( 

233 f"FluentWorkflow cannot select files for the reader " 

234 f"{self._reader_name!r}; use one of {sorted(_READER_FILE_FORMATS)}" 

235 ) 

236 raise ValueError(msg) 

237 site_cfg = load_config().sites.sites[self._site.name] 

238 day_files = _day_files( 

239 self._site.name, site_cfg, _resolve_date(date), reader_format 

240 ) 

241 

242 for name in receiver_list: 

243 files = day_files[name] 

244 if not files: 

245 log.warning("no_files", receiver=name) 

246 continue 

247 log.info("files_discovered", receiver=name, n_files=len(files)) 

248 

249 datasets_for_recv = [] 

250 for file in files: 

251 reader_obj = ReaderFactory.create(self._reader_name, fpath=file.path) 

252 ds = reader_obj.to_ds(write_global_attrs=True) 

253 # As a run does: the store records the reader's format. 

254 ds.attrs[SOURCE_FORMAT_ATTR] = reader_obj.source_format 

255 

256 # Filter variables 

257 if self._keep_vars: 

258 drop = [v for v in ds.data_vars if v not in set(self._keep_vars)] 

259 if drop: 

260 ds = ds.drop_vars(drop) 

261 

262 datasets_for_recv.append(ds) 

263 

264 if datasets_for_recv: 

265 self._datasets[name] = _concat_epoch_datasets(datasets_for_recv) 

266 log.info("read_complete", receiver=name, files=len(datasets_for_recv)) 

267 

268 return self # never reached (decorator returns self), but aids type checkers

270 @step 

271 def preprocess(self, agency: str = "COD") -> FluentWorkflow: 

272 """Apply auxiliary preprocessing to loaded datasets. 

273 

274 Parameters 

275 ---------- 

276 agency : str 

277 Analysis center for auxiliary products (default ``"COD"``). 

278 """ 

279 log = self.log.bind(agency=agency) 

280 

281 for name, ds in self._datasets.items(): 

282 try: 

283 from canvod.auxiliary import preprocess_aux_for_interpolation 

284 

285 ds = preprocess_aux_for_interpolation(ds) 

286 self._datasets[name] = ds 

287 log.info("preprocess_complete", receiver=name) 

288 except ImportError: 

289 log.debug("canvod.auxiliary not available, skipping preprocessing") 

290 

291 return self

293 @step 

294 def augment( 

295 self, 

296 source: str = "final", 

297 agency: str = "COD", 

298 date: str | None = None, 

299 ) -> FluentWorkflow: 

300 """Augment loaded datasets with theta/phi from ephemeris data. 

301 

302 Parameters 

303 ---------- 

304 source : str 

305 Ephemeris source: ``"final"`` (SP3/CLK), ``"rapid"``, 

306 ``"broadcast"`` (SBF only). 

307 agency : str 

308 Analysis center code (default ``"COD"``). 

309 date : str, optional 

310 Date in ``YYYYDOY`` format. If not provided, inferred from 

311 the most recent ``.read()`` call. 

312 """ 

313 from canvod.auxiliary.ephemeris.provider import ( 

314 AgencyEphemerisProvider, 

315 ) 

316 from canvod.config import load_config 

317 

318 log = self.log.bind(source=source, agency=agency) 

319 

320 config = load_config() 

321 site_cfg = config.sites.sites[self._site.name] 

322 

323 if source in ("final", "rapid"): 

324 provider = AgencyEphemerisProvider( 

325 agency=agency, 

326 product_type=source, 

327 ) 

328 _date = date or self._last_date 

329 if _date: 

330 provider.preprocess_day(_date, site_cfg) 

331 

332 from canvod.auxiliary.position.position import ECEFPosition 

333 

334 for name, ds in self._datasets.items(): 

335 try: 

336 rx_pos = ECEFPosition.from_ds_metadata(ds) 

337 except (KeyError, ValueError): # fmt: skip 

338 log.warning( 

339 "no_receiver_position", 

340 receiver=name, 

341 ) 

342 continue 

343 self._datasets[name] = provider.augment_dataset( 

344 ds, 

345 rx_pos, 

346 ) 

347 log.info("augment_complete", receiver=name) 

348 else: 

349 log.warning("augment_source_not_supported", source=source) 

350 

351 return self

353 @step 

354 def grid(self, kind: str | None = None, **params: Any) -> FluentWorkflow: 

355 """Build a hemisphere grid and assign cell IDs to all datasets. 

356 

357 Parameters 

358 ---------- 

359 kind : str, optional 

360 Grid type override. Defaults to the value set at init. 

361 **params 

362 Passed to :meth:`GridFactory.create` (e.g. 

363 ``angular_resolution=5.0``). 

364 """ 

365 from canvod.grids import add_cell_ids_to_ds_fast 

366 

367 grid_type = kind or self._grid_type 

368 builder = GridFactory.create(grid_type, **params) 

369 self._grid = builder.build() 

370 

371 for name, ds in self._datasets.items(): 

372 self._datasets[name] = add_cell_ids_to_ds_fast(ds, self._grid, grid_type) 

373 

374 self.log.info("grid_complete", grid=grid_type, ncells=self._grid.ncells) 

375 return self

377 @step 

378 def vod(self, canopy: str, reference: str) -> FluentWorkflow: 

379 """Compute vegetation optical depth for a receiver pair. 

380 

381 Parameters 

382 ---------- 

383 canopy : str 

384 Canopy receiver name (e.g. ``"canopy_01"``). 

385 reference : str 

386 Sky/reference receiver name (e.g. ``"reference_01"``). 

387 """ 

388 canopy_ds = self._datasets[canopy] 

389 ref_ds = self._datasets[reference] 

390 

391 calculator = VODFactory.create( 

392 self._vod_calculator_name, 

393 canopy_ds=canopy_ds, 

394 sky_ds=ref_ds, 

395 ) 

396 self._vod_result = calculator.calculate_vod() 

397 self._vod_pair = (canopy, reference) 

398 

399 self.log.info("vod_complete", canopy=canopy, reference=reference) 

400 return self

417 @terminal 

418 def to_store(self) -> FluentWorkflow: 

419 """Execute the plan and write results to Icechunk storage. 

420 

421 Takes the same step first as ``canvodpy run``: a new store gets 

422 its metadata from the settings, and receivers or VOD analyses the 

423 store's metadata doesn't list are refused. VOD goes to the group 

424 ``{calculator}/{analysis}`` of the VOD analysis of the settings that 

425 pairs the ``.vod()`` receivers, as ``site.vod`` writes it. 

426 

427 Raises 

428 ------ 

429 ValueError 

430 If no VOD analysis of the settings pairs the ``.vod()`` 

431 receivers, a dataset does not name the format of its reader, 

432 or the store refuses the write (names the command to run). 

433 """ 

434 research_site = self._site._site 

435 if self._vod_result is not None: 

436 analysis = self._vod_analysis() 

437 self.log.info("to_store_vod", analysis=analysis) 

438 research_site.ensure_store_metadata( 

439 "vod_store", [analysis], self._vod_calculator_name 

440 ) 

441 self._site.vod_store.write_or_append_group( 

442 self._vod_result, f"{self._vod_calculator_name}/{analysis}" 

443 ) 

444 else: 

445 from canvodpy.orchestrator.processor import _source_format_of 

446 

447 receivers = list(self._datasets) 

448 research_site.ensure_store_metadata( 

449 "gnss_store", 

450 receivers, 

451 _source_format_of(self._datasets.values(), receivers), 

452 ) 

453 for name, ds in self._datasets.items(): 

454 self.log.info("to_store_dataset", receiver=name) 

455 self._site.gnss_store.write_or_append_group(ds, name) 

456 return self

458 def _vod_analysis(self) -> str: 

459 """The VOD analysis of the settings that pairs the .vod() receivers.""" 

460 canopy, reference = self._vod_pair or (None, None) 

461 for name, analysis in self._site.vod_analyses.items(): 

462 if (analysis.canopy_receiver, analysis.reference_receiver) == ( 

463 canopy, 

464 reference, 

465 ): 

466 return name 

467 msg = ( 

468 f"No VOD analysis in the settings pairs {canopy!r} with " 

469 f"{reference!r} (sites.{self._site.name}.vod_analyses); add one " 

470 "to store this VOD." 

471 ) 

472 raise ValueError(msg)
