1"""Fingerprinting for the network-wide cache of interpolated orbits and clocks. 

2 

3The broadcast ephemeris of navigation files shares the cache with its own 

4fingerprint (:func:`compute_broadcast_cache_fingerprint`). 

5 

6The aux cache is keyed by a fingerprint that is 

7deliberately site-independent -- ephemeris products are satellite-based, not 

8site-based, so the same fingerprint should hit for every site sharing the 

9same agency/product/ephemeris-source config on the same date, regardless of 

10which site's pipeline run happens to populate it first. 

11""" 

12 

13from __future__ import annotations 

14 

15import hashlib 

16import json 

17from pathlib import Path 

18 

19# Target epoch grid spacing for the shared cache, independent of any single 

20# site's own sampling interval (finer than any real deployment's rate, so 

21# the consumer's nearest-neighbor `.sel()` always lands close). Included in 

22# the fingerprint, so bumping it later orphans old entries instead of 

23# silently misreading them against a coarser grid. 

24CANONICAL_AUX_GRID_SECONDS = 1.0 

25 

26# Version of the cached content. Bump it when the computed values change for 

27# the same input files, so entries written by older code are not reused. 

28# 2: CLK clock bias read in seconds (was divided by 1e6). 

29# 3: one entry per satellite (sv), not per signal (#573). 

30# 4: orbits interpolated with a polynomial of degree 10, not a cubic 

31# Hermite spline (#563). 

32AUX_CACHE_FORMAT_VERSION = 4 

33 

34 

35def compute_aux_cache_fingerprint( 

36 *, 

37 agency: str, 

38 product_type: str, 

39 ephemeris_source: str, 

40 canonical_grid_seconds: float, 

41 source_file_paths: dict[str, Path | None], 

42 aggregate_glonass_fdma: bool, 

43) -> str: 

44 """Compute a stable fingerprint for one day's aux cache entry. 

45 

46 Deliberately excludes anything site-specific (``keep_sids``, 

47 ``keep_gnss_observables``, ``sampling_interval``, site identity) -- 

48 consumers already filter the cached aux data down to their own SID/epoch 

49 needs at read time (see ``preprocess_with_hermite_aux`` step 2-3), so a 

50 broader/denser shared cache entry is always a valid superset. 

51 

52 ``aggregate_glonass_fdma`` is included because it changes the GLONASS 

53 signal IDs of the cached data. 

54 

55 Source file mtimes are included so a silent upstream reprocessing (e.g. 

56 rapid orbit revised to final) produces a new fingerprint rather than 

57 reusing a stale cache entry. 

58 """ 

59 sections = { 

60 "format_version": AUX_CACHE_FORMAT_VERSION, 

61 "agency": agency, 

62 "product_type": product_type, 

63 "ephemeris_source": ephemeris_source, 

64 "canonical_grid_seconds": canonical_grid_seconds, 

65 "aggregate_glonass_fdma": aggregate_glonass_fdma, 

66 "source_files": { 

67 name: ( 

68 {"path": str(path), "mtime": path.stat().st_mtime} 

69 if path is not None and path.exists() 

70 else None 

71 ) 

72 for name, path in sorted(source_file_paths.items()) 

73 }, 

74 } 

75 return hashlib.sha256( 

76 json.dumps(sections, sort_keys=True, default=str).encode() 

77 ).hexdigest() 

78 

79 

80def compute_broadcast_cache_fingerprint( 

81 *, 

82 nav_files: list[Path], 

83 canonical_grid_seconds: float, 

84 aggregate_glonass_fdma: bool, 

85) -> str: 

86 """Compute a stable fingerprint for one day's broadcast aux cache entry. 

87 

88 The broadcast counterpart of :func:`compute_aux_cache_fingerprint`: 

89 the orbits come from a receiver's navigation files, so the fingerprint 

90 holds the SHA-256 of each file's content instead of an agency product. 

91 A changed or added navigation file gives a new fingerprint; receivers 

92 and sites with the same navigation files share the entry. The section 

93 ``ephemeris_source`` keeps it apart from every precise entry. 

94 """ 

95 sections = { 

96 "format_version": AUX_CACHE_FORMAT_VERSION, 

97 "ephemeris_source": "broadcast", 

98 "canonical_grid_seconds": canonical_grid_seconds, 

99 "aggregate_glonass_fdma": aggregate_glonass_fdma, 

100 "nav_files": sorted( 

101 hashlib.sha256(path.read_bytes()).hexdigest() for path in nav_files 

102 ), 

103 } 

104 return hashlib.sha256( 

105 json.dumps(sections, sort_keys=True, default=str).encode() 

106 ).hexdigest()
